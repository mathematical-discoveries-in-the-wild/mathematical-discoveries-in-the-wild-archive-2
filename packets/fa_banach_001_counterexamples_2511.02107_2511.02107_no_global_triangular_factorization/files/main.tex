\title{A finite-rank obstruction to global triangular\\
factorization in \(GL^p(\mathcal H)\)}
\author{Candidate counterexample packet for arXiv:2511.02107}
\date{17 August 2026}

\begin{document}
\maketitle

\begin{abstract}
For every \(1<p<\infty\), the multiplication map underlying the
upper/lower triangular factorization in Goli\'nski--Tumpach Remark~7.4 is not
surjective on \(GL^p(\mathcal H)\).  A rotation through \(\pi/2\) on the first
two basis vectors, extended by the identity, is an explicit counterexample.
It lies in every Schatten restricted group and even in its identity
component.  Nearby rotations admit explicit triangular factorizations, whose
factors blow up at the exceptional angle; this pinpoints why the local
factorization in Proposition~7.3 cannot be globalized.
\end{abstract}

\paragraph{Status and scope.}
This is a candidate full negative answer to the global triangular
factorization question in source Remark~7.4.  It does not address the
different Iwasawa-decomposition question for restricted groups on polarized
Hilbert spaces in source Remark~6.6.

\section{Setting}

Let \(\mathcal H\) be a complex separable Hilbert space with orthonormal basis
\((e_n)_{n\geq1}\), and let \(\mathcal S^p\) be the Schatten ideal.  The
source writes
\[
 GL^p(\Hh)=(\Id+\Sp)\cap GL(\Hh).
\]
Its strictly upper and lower-plus-diagonal Lie algebras are
\begin{align*}
 \Sp_{++}&=\{X\in\Sp:Xe_n\in\operatorname{span}(e_1,\ldots,e_{n-1})\},\\
 \Sp_-&=\{Y\in\Sp:Ye_n\in\clspan(e_n,e_{n+1},\ldots)\}.
\end{align*}
Thus the factorization used in source Proposition~7.3 has the form
\begin{equation}\label{eq:factor}
 A=g_+^{-1}g_-,\qquad
 g_+\in G_+:=(\Id+\Sp_{++})\cap GL(\Hh),\quad
 g_-\in G_-:=(\Id+\Sp_-)\cap GL(\Hh).
\end{equation}
It exists locally near the identity.  Remark~7.4 asks whether it exists
globally on \(GL^p(\Hh)\).

\section{Proof intuition}

The first vector is rigid under every unit upper-triangular factor:
\(g_+e_1=e_1\).  The second vector is constrained under every lower
triangular factor: \(g_-e_2\) cannot have an \(e_1\)-component.  An operator
that sends \(e_2\) to \(-e_1\) makes these two requirements collide
immediately.  Extending this two-dimensional operator by the identity keeps
it inside every Schatten restricted group, so infinite tail coordinates
cannot supply a hidden pivot.

\section{Counterexample and sharp path}

\begin{theorem}\label{thm:main}
For every \(1<p<\infty\), the map
\[
 G_+\times G_-\longrightarrow GL^p(\Hh),\qquad
 (g_+,g_-)\longmapsto g_+^{-1}g_-
\]
is not surjective.  This remains false if the upper factor is allowed an
arbitrary invertible diagonal.
\end{theorem}

\begin{proof}
Let \(A\) act on \(E=\operatorname{span}(e_1,e_2)\) by
\[
 A|_E=\begin{pmatrix}0&-1\\1&0\end{pmatrix}
\]
in the ordered basis \((e_1,e_2)\), and let \(A\) be the identity on
\(E^\perp\).  It is unitary and \(A-\Id\) has rank two.  Therefore
\(A\in GL^p(\Hh)\) for every finite \(p\).

Suppose \eqref{eq:factor} held.  Then \(g_-=g_+A\).  Strict upper
triangularity gives \(g_+e_1=e_1\), whereas \(Ae_2=-e_1\).  Consequently
\begin{equation}\label{eq:collision}
 g_-e_2=g_+Ae_2=-g_+e_1=-e_1.
\end{equation}
But by lower triangularity,
\[
 g_-e_2\in\clspan(e_2,e_3,\ldots),
\]
contradicting \eqref{eq:collision}.

If the upper triangular group is enlarged to allow arbitrary invertible
diagonal entries, then \(g_+e_1=c e_1\) for some \(c\neq0\), and the same
argument gives \(g_-e_2=-c e_1\), again impossible.
\end{proof}

The counterexample is not separated from the identity by a component
obstruction.  Define the smooth finite-rank rotation path
\begin{equation}\label{eq:path}
 A_\theta|_E=
 \begin{pmatrix}\cos\theta&-\sin\theta\\
 \sin\theta&\cos\theta\end{pmatrix},
 \qquad A_\theta|_{E^\perp}=\Id.
\end{equation}

\begin{theorem}\label{thm:path}
If \(\cos\theta\neq0\), then \(A_\theta\) has a factorization of the form
\eqref{eq:factor}.  One may take, on \(E\),
\begin{equation}\label{eq:explicit}
 g_+(\theta)=\begin{pmatrix}1&\tan\theta\\0&1\end{pmatrix},
 \qquad
 g_-(\theta)=\begin{pmatrix}\sec\theta&0\\
 \sin\theta&\cos\theta\end{pmatrix},
\end{equation}
with both operators equal to the identity on \(E^\perp\).  At
\(\theta=\pi/2\) no such factorization exists, and the displayed factors
blow up as \(\theta\to\pi/2\).
\end{theorem}

\begin{proof}
Both factors in \eqref{eq:explicit} are invertible finite-rank perturbations
of identity and have the required triangular forms.  Direct multiplication
using \(\sin^2\theta+\cos^2\theta=1\) gives
\[
 g_+(\theta)A_\theta=g_-(\theta),
\]
hence \(A_\theta=g_+(\theta)^{-1}g_-(\theta)\).  Moreover
\(\|g_+(\theta)-\Id\|_p=|\tan\theta|\), while the upper-left diagonal entry
of \(g_-(\theta)\) is \(\sec\theta\).  Both are unbounded at \(\pi/2\), and
Theorem~\ref{thm:main} proves nonexistence at the endpoint.
\end{proof}

\begin{corollary}
The local factorization chart containing the identity cannot be continued
globally along the path \(A_\theta\), even though that path consists entirely
of finite-rank perturbations of identity.
\end{corollary}

\paragraph{Convention check.}
The same endpoint also fails a factorization in the reverse order
\(A=LU\) with invertible lower and upper triangular factors.  Indeed,
\(Ue_1=c e_1\) with \(c\neq0\) would force \(Le_1=c^{-1}e_2\).  Lower
triangularity then makes every entry in the first row of \(L\) zero, so
\(L\) is not surjective.  The negative answer is therefore independent of
whether the phrase ``LU factorization'' is attached to the source's displayed
upper--lower order or to the reverse naming convention.
