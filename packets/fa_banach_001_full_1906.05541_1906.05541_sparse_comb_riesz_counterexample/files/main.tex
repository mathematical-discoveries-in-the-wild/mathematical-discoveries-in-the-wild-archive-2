\begin{abstract}
We give a claimed affirmative answer to Open Problem 1.6 in Spector's
\emph{A Noninequality for the Fractional Gradient}.  For every $d\geq2$ and
$0<\beta<1$, we construct a nonnegative
$u\in L^1(\mathcal H^{d-\beta}_\infty)$ such that
\[
 \sup_{t>0}t\,\mathcal H^{d-\beta}_\infty
       (\{\lvert DI_1u\rvert>t\})=+\infty.
\]
The construction uses sparse one-dimensional combs, a fixed tangential cube,
and a summable gluing of widely separated smooth blocks.
\end{abstract}

\section*{Source Question}

Spector asks on page 6, Open Problem 1.6 of \cite{Spector}:
\begin{quote}
For $\beta\in(0,1)$, can one find
$u\in L^1(\mathcal H^{d-\beta}_\infty)$ such that
\[
 \sup_{t>0}t\,\mathcal H^{d-\beta}_\infty
       (\{\lvert DI_1u\rvert>t\})=+\infty?
\]
\end{quote}

\begin{center}
\includegraphics[width=0.91\textwidth]{figures/open_problem_crop.png}
\end{center}

For $0<s<d$, we use the normalization
\[
 \Hs{s}(E)=\inf\left\{\sum_i\omega_s r_i^s:
 E\subseteq\bigcup_i B(x_i,r_i)\right\}.
\]
The harmless factor $\omega_s$ will always be absorbed into constants.  The
Choquet functional is
\[
 \choq f=\int_0^\infty \Hs{s}(\{\lvert f\rvert>t\})\,dt,
\]
and the space in the question is the completion of $C_c(\R^d)$ for this
functional, as in \cite{Spector}.

\section*{Proof Intuition}

A single cube has a logarithmic Riesz singularity near one hypersurface.  If
$s=d-\beta>d-1$, a slab of thickness $\delta$ has $s$-content roughly
$\delta^{1-\beta}$, so that one logarithmic singularity is too thin to force
divergence.

Instead, place $N$ very short intervals at spacing $1/N$.  Immediately to the
right of every interval the adjacent endpoint contributes $\asymp\log N$ to
the Hilbert kernel, while all nonadjacent intervals contribute $o(1)$.  The
test gaps can be chosen so that their union carries a probability measure
with uniform $(1-\beta)$-Frostman growth.  Tensoring with a fixed
$(d-1)$-dimensional cube lifts this to uniformly positive
$(d-\beta)$-content, and integration of the last Riesz kernel in the
tangential variables produces the Hilbert kernel $1/h$ up to a bounded
error.

Each finite comb has bounded Choquet cost but Riesz height $\asymp\log N$.
Taking rapidly increasing $N$, summably decreasing amplitudes, and widely
separated translates gives one function with finite Choquet norm and
unbounded weak Riesz quasi-norm.

\section*{The Result}

\begin{theorem}\label{thm:main}
Let $d\geq2$ and $0<\beta<1$.  Put $s=d-\beta$.  There exists a nonnegative
function
\[
 u\in L^1(\Hs{s})\cap L^1(\R^d)
\]
such that
\[
 \sup_{t>0}t\,\Hs{s}(\{x:\lvert DI_1u(x)\rvert>t\})=+\infty.
\]
Moreover, $u$ can be represented as a locally finite sum of nonnegative
$C_c^\infty$ functions.
\end{theorem}

\begin{proof}
Set
\[
 q=1-\beta\in(0,1),\qquad s=(d-1)+q.
\]
Choose real numbers
\begin{equation}\label{eq:parameters}
 1<r<p<\frac1q.
\end{equation}

\medskip
\noindent\textbf{Step 1: a sparse one-dimensional comb.}
For a sufficiently large integer $N$, let
\[
 \ell_N=N^{-r},\qquad \delta_N=N^{-p},
\]
and define
\begin{align*}
 E_N&=\bigcup_{k=0}^{N-1}
       \left[\frac{k}{N},\frac{k}{N}+\ell_N\right],\\
 J_N&=\bigcup_{k=0}^{N-1}J_{k,N},\\
 J_{k,N}&=\left[\frac{k}{N}+\ell_N+\delta_N,
                   \frac{k}{N}+\ell_N+2\delta_N\right].
\end{align*}
As $r,p>1$, these intervals are disjoint and contained in $[0,1]$ once
$N$ is large.

We claim that, uniformly for $z\in J_N$,
\begin{equation}\label{eq:hilbert-lower}
 \left\lvert\int_{E_N}\frac{dy}{z-y}\right\rvert
 \geq c\log N.
\end{equation}
Indeed, let $z\in J_{k,N}$ and write
$z=k/N+\ell_N+\theta$ with
$\delta_N\leq\theta\leq2\delta_N$.  The $k$th component contributes
\[
 \int_{k/N}^{k/N+\ell_N}\frac{dy}{z-y}
 =\log\frac{\ell_N+\theta}{\theta}
 \geq (p-r)\log N-\log2.
\]
For $j\neq k$, the distance from $z$ to the $j$th component is at least
$c\lvert j-k\rvert/N$.  Therefore the absolute value of the sum of all
nonadjacent contributions is at most
\[
 C\sum_{m=1}^{N}\frac{N\ell_N}{m}
 \leq C N^{1-r}\log N=o(1).
\]
This proves \eqref{eq:hilbert-lower}.

\medskip
\noindent\textbf{Step 2: the target gaps have uniformly positive content.}
Let $\nu_N$ be normalized Lebesgue measure on $J_N$; thus it has density
$(N\delta_N)^{-1}$ on $J_N$ and total mass one.  We show that
\begin{equation}\label{eq:frostman-one-d}
 \nu_N(I)\leq C\lvert I\rvert^q
\end{equation}
for every interval $I\subset\R$, with $C$ independent of $N$.

Write $\rho=\lvert I\rvert$.  If $\rho\leq\delta_N$, then
\[
 \nu_N(I)\leq\frac{\rho}{N\delta_N}
 \leq \rho^q\frac{1}{N\delta_N^q}
 \leq\rho^q,
\]
because $N\delta_N^q=N^{1-pq}\geq1$.  If
$\delta_N\leq\rho\leq1/N$, then $I$ meets at most two components of $J_N$,
and
\[
 \nu_N(I)\leq\frac2N\leq2\delta_N^q\leq2\rho^q.
\]
If $1/N\leq\rho\leq1$, then $I$ meets at most $C(N\rho+1)$ components, so
$\nu_N(I)\leq C\rho\leq C\rho^q$.  The case $\rho\geq1$ is immediate.
This proves \eqref{eq:frostman-one-d}.

Let
\[
 A=[-2,2]^{d-1},\qquad A_0=[-1,1]^{d-1},\qquad
 B_N=A\times E_N,\qquad F_N=A_0\times J_N.
\]
Take normalized $(d-1)$-dimensional Lebesgue measure $\lambda$ on $A_0$.
For every ball $B(x,\rho)\subset\R^d$, the product probability measure
$\mu_N=\lambda\otimes\nu_N$, supported on $F_N$, satisfies
\[
 \mu_N(B(x,\rho))\leq C\rho^{d-1}\rho^q=C\rho^s
 \quad(0<\rho\leq1),
\]
and the same estimate is trivial for $\rho\geq1$.  Consequently, for any
ball cover $F_N\subseteq\bigcup_iB(x_i,r_i)$,
\[
 1=\mu_N(F_N)\leq\sum_i\mu_N(B(x_i,r_i))
 \leq C\sum_i r_i^s.
\]
Taking the infimum over covers gives the uniform lower bound
\begin{equation}\label{eq:content-lower}
 \Hs{s}(F_N)\geq c>0.
\end{equation}

\medskip
\noindent\textbf{Step 3: lift the Hilbert lower bound to a Riesz component.}
Up to a fixed nonzero dimensional constant, the last component of $DI_1$ is
\[
 R_df(x)=c_d\,\operatorname{p.v.}\int_{\R^d}
 \frac{x_d-y_d}{\lvert x-y\rvert^{d+1}}f(y)\,dy.
\]
For $x'\in A_0$ and $0<\lvert h\rvert\leq1$, set
\[
 K_{x'}(h)=c_d\int_A
 \frac{h}{(\lvert x'-y'\rvert^2+h^2)^{(d+1)/2}}\,dy'.
\]
Integration over all of $\R^{d-1}$ and the change of variables
$w=\lvert h\rvert z$ give
\[
 c_d\int_{\R^{d-1}}
 \frac{h}{(\lvert w\rvert^2+h^2)^{(d+1)/2}}\,dw
 =\frac{\kappa_d}{h}
\]
for a nonzero constant $\kappa_d$.  Since $x'\in A_0$, the translated set
$x'-A$ contains the unit ball.  Hence the omitted tail satisfies
\[
 \left\lvert K_{x'}(h)-\frac{\kappa_d}{h}\right\rvert
 \leq C\lvert h\rvert
 \int_{\lvert w\rvert\geq1}\lvert w\rvert^{-d-1}\,dw
 \leq C\lvert h\rvert.
\]
For $(x',z)\in F_N$, no principal value is needed and Fubini's theorem gives
\begin{align*}
 R_d\1_{B_N}(x',z)
 &=\int_{E_N}K_{x'}(z-y_d)\,dy_d\\
 &=\kappa_d\int_{E_N}\frac{dy_d}{z-y_d}+O(1),
\end{align*}
where the error is uniform because $z,y_d\in[0,1]$ and
$\lvert E_N\rvert=N\ell_N\leq1$.  By \eqref{eq:hilbert-lower},
\begin{equation}\label{eq:riesz-block}
 \lvert R_d\1_{B_N}(x)\rvert\geq c\log N-C
 \qquad(x\in F_N).
\end{equation}

We now smooth each block.  Because
$\operatorname{dist}(F_N,B_N)=\delta_N$, choose a standard mollifier at a
scale $\varepsilon_N<\delta_N/4$ so small that
\[
 v_N=\1_{B_N}*\rho_{\varepsilon_N}
\quad\text{satisfies}\quad
 \sup_{x\in F_N}\lvert R_d(v_N-\1_{B_N})(x)\rvert\leq1.
\]
This is possible because $v_N\to\1_{B_N}$ in ordinary $L^1$, while the
Riesz kernel is uniformly bounded on pairs consisting of a point of $F_N$
and a point in the $\varepsilon_N$-neighborhood of $B_N$.  Thus
$v_N\in C_c^\infty$, $0\leq v_N\leq1$, its support lies in a fixed bounded
box independent of $N$, and
\begin{equation}\label{eq:smooth-block}
 \lvert R_dv_N(x)\rvert\geq c\log N-C-1
 \qquad(x\in F_N).
\end{equation}

\medskip
\noindent\textbf{Step 4: glue the finite witnesses into one function.}
Let
\[
 a_n=2^{-n},\qquad N_n=\left\lceil e^{4^n}\right\rceil,
 \qquad R_n=L4^n,
\]
where $L$ will be chosen large.  Increasing finitely many $N_n$ if needed,
assume all preceding estimates apply.  Translate in the last coordinate:
\begin{align*}
 \widetilde v_n(x',x_d)&=v_{N_n}(x',x_d-R_n),\\
 \widetilde F_n&=F_{N_n}+(0,R_n).
\end{align*}
For large $L$, the supports of the $\widetilde v_n$ are pairwise disjoint.
Define
\[
 u=\sum_{n=1}^{\infty}a_n\widetilde v_n.
\]

First, $u$ belongs to the precise Choquet space used in the question.  The
partial sums are in $C_c^\infty(\R^d)$.  Since the supports are disjoint and
every $\operatorname{supp}\widetilde v_n$ is contained in a translate of one
fixed box, countable subadditivity of Hausdorff content yields, for $M'>M$,
\begin{align*}
 &\int_0^\infty\Hs{s}
 \left(\left\{\left\lvert\sum_{n=M+1}^{M'}a_n\widetilde v_n\right\rvert
 >t\right\}\right)dt\\
 &\qquad\leq\sum_{n=M+1}^{M'}
 \int_0^\infty\Hs{s}(\{a_n\widetilde v_n>t\})\,dt
 \leq C\sum_{n=M+1}^{M'}a_n\longrightarrow0.
\end{align*}
Thus the partial sums are Cauchy in the defining functional, and their
pointwise, locally finite sum $u$ represents an element of $L^1(\Hs{s})$.
Also
$\lVert u\rVert_{L^1(\R^d)}\leq C\sum_na_n<\infty$.

It remains to control interactions between different blocks.  For
$x\in\widetilde F_n$ and $m\neq n$, the separation $R_j=L4^j$, the kernel
bound $\lvert K(x-y)\rvert\leq C\lvert x-y\rvert^{-d}$, and
$\lVert\widetilde v_m\rVert_1\leq C$ give
\[
 \sum_{m\neq n}a_m\lvert R_d\widetilde v_m(x)\rvert
 \leq C L^{-d}.
\]
Indeed, all $m<n$ blocks are at distance comparable to $R_n$, while each
$m>n$ block is at distance comparable to $R_m$; summing the two geometric
tails proves the displayed estimate.  Choose $L$ so large that this bound is
at most one.

Using \eqref{eq:smooth-block} and $\log N_n\geq4^n$, we obtain, uniformly on
$\widetilde F_n$ and for all sufficiently large $n$,
\[
 \lvert R_du(x)\rvert
 \geq a_n(c\log N_n-C-1)-1
 \geq c_1 2^n.
\]
Since the norm of the vector Riesz transform dominates the absolute value of
its last component, if $t_n=c_1 2^{n-1}$ then
\[
 \widetilde F_n\subseteq\{\lvert DI_1u\rvert>t_n\}.
\]
Hausdorff content is translation invariant, so \eqref{eq:content-lower}
implies
\[
 t_n\Hs{s}(\{\lvert DI_1u\rvert>t_n\})
 \geq t_n\Hs{s}(\widetilde F_n)
 \geq c t_n\longrightarrow\infty.
\]
This proves the theorem.
\end{proof}

\section*{Verification Notes}

The proof is entirely quantitative.  Its three threshold inequalities are
$r>1$, which makes the nonadjacent comb contribution vanish;
$p>r$, which creates logarithmic height; and $pq<1$, which ensures
$N\delta_N^q\geq1$ and hence positive target content.  Such parameters exist
exactly because $q=1-\beta<1$.

The auxiliary script \texttt{code/check\_comb\_estimates.py} evaluates the
exact finite sum
\[
 \sum_k\log\frac{|z-k/N|}{|z-k/N-\ell_N|}
\]
on every target interval midpoint and samples the Frostman quotient
$\nu_N(I)/|I|^q$ for representative values of $q$.  This is a sanity check,
not an input to the proof.

The constructed function is not compactly supported.  This is permitted by
the open problem, and its membership in the completion of $C_c$ is verified
directly by the Choquet-tail estimate above.  The argument gives every
$d\geq2$ and every $0<\beta<1$, with no endpoint claim at $\beta=0$ or
$\beta=1$.
