\title{Removing the full-preimage sigma-algebra hypothesis:\\
an exact transfer-operator characterization of\\
$k$-quasi $n$-power posinormal weighted composition operators}
\author{Candidate full solution packet for arXiv:2507.06511}
\date{12 August 2026}

\begin{document}
\maketitle

\begin{abstract}
The first future-work direction in arXiv:2507.06511 asks for the assumption
$\varphi^{-j}(\Sigma)=\Sigma$ to be relaxed in the composition and weighted
composition results.  We remove it completely.  For an arbitrary nonsingular
transformation inducing a bounded composition operator, the Perron transfer
operators $\Perr_j=(C_\varphi^*)^j$ give exact necessary-and-sufficient
conditions.  For the weighted composition operator itself, the conditions are
scalar almost-everywhere inequalities in both regimes $k\ge n$ and $k<n$; the
latter follows from a sharp weighted transfer lemma.  For the adjoint, the
condition is an exact one-step compressed transfer inequality.  The results
allow noninvertible transformations and complex weights and specialize to the
unweighted case by setting the weight equal to one.
\end{abstract}

The sigma-algebra condition and measure preservation are logically distinct:
the former is a full-preimage/bimeasurability condition.  The theorem below
removes precisely that sigma-algebra hypothesis while retaining the source's
standard nonsingularity and boundedness framework.  It covers both ordinary
and weighted composition operators, as well as their adjoints, which are the
subjects of Sections~2--3 of the source.

\section{Notation and transfer facts}

Let $(X,\Sigma,\mu)$ be a complete sigma-finite measure space, let
$\varphi:X\to X$ be nonsingular, and suppose
\[
 C=C_\varphi:L^2(\mu)\to L^2(\mu),\qquad Cf=f\circ\varphi,
\]
is bounded.  Let $\pi\in L^\infty(\mu)$ and
$W=M_\pi C$.  No assumption $\varphi^{-j}(\Sigma)=\Sigma$ is made.

For $j\ge0$, put
\[
 \pi_0=1,\qquad
 \pi_j=\prod_{i=0}^{j-1}\pi\circ\varphi^i,
 \qquad \Perr_j=(C^*)^j.
\]
On $L^2$, $\Perr_j$ is the adjoint transfer operator.  On nonnegative
measurable functions it has its usual monotone extension, characterized by
\[
 \int_X \Perr_j(a)\,\overline g\,d\mu
   =\int_X a\,\overline{g\circ\varphi^j}\,d\mu
 \tag{2.1}
\]
whenever the two sides are defined.  In particular,
\[
 \Perr_j\bigl(a(g\circ\varphi^j)\bigr)=g\Perr_j(a),
 \qquad
 \int_X\Perr_j(a)\,d\mu=\int_Xa\,d\mu.
 \tag{2.2}
\]
Define the nonnegative functions
\[
 q_j:=\Perr_j(|\pi_j|^2),\qquad q_0=1.
 \tag{2.3}
\]
Thus $q_j$ is the Radon--Nikodym density of
$A\mapsto\int_{\varphi^{-j}(A)}|\pi_j|^2d\mu$.

We use the extended quotient convention
\[
 \frac{|b|^2}{q}=0\quad\hbox{on }\{b=0=q\},
 \qquad
 \frac{|b|^2}{q}=+\infty\quad\hbox{on }\{q=0,\ b\ne0\}.
 \tag{2.4}
\]

The following sharp lemma is the mechanism that turns the noninvertible
$k<n$ case into a scalar condition.

\begin{lemma}[sharp weighted transfer bound]\label{lem:transfer}
Let $r\ge0$, let $q\ge0$, and let $b$ be measurable, with the displayed
expressions well-defined by truncation.  For $\lambda\ge0$, the following are
equivalent:
\begin{align*}
 \|\Perr_r(bf)\|_2^2
   &\le \lambda^2\int_X q|f|^2d\mu
      &&\text{for every admissible }f, \tag{2.5}\\
 \Perr_r\left(\frac{|b|^2}{q}\right)
   &\le\lambda^2
      &&\text{almost everywhere}. \tag{2.6}
\end{align*}
Consequently, the optimal squared constant in (2.5) is
$\|\Perr_r(|b|^2/q)\|_\infty$, possibly $+\infty$.
\end{lemma}

\begin{proof}
The transfer Cauchy--Schwarz inequality and (2.2) give
\[
 |\Perr_r(bf)|^2
 \le \Perr_r\left(\frac{|b|^2}{q}\right)\Perr_r(q|f|^2).
\]
Thus (2.6), integration, and (2.2) imply (2.5).

Conversely, write $a=|b|^2/q$.  For bounded $g$ of finite-measure support,
first on truncation sets where all quantities are bounded, take
\[
 f=\frac{\overline b}{q}(g\circ\varphi^r).
\]
Then (2.2) yields
\[
 \Perr_r(bf)=g\Perr_r(a),\qquad
 \int_Xq|f|^2d\mu=\int_X|g|^2\Perr_r(a)d\mu.
\]
Substitution in (2.5) gives
\[
 \int_X|g|^2\Perr_r(a)^2d\mu
 \le\lambda^2\int_X|g|^2\Perr_r(a)d\mu.
\]
Arbitrary such $g$ force $\Perr_r(a)\le\lambda^2$ almost everywhere.
Monotone truncation proves the assertion with the convention (2.4).
\end{proof}

\section{Exact characterization without the sigma-algebra hypothesis}

Recall that an operator $A$ is $k$-quasi $n$-power posinormal if some
$\lambda>0$ satisfies
\[
 A^{*k}(\lambda^2A^*A-A^nA^{*n})A^k\ge0.
 \tag{3.1}
\]

\begin{theorem}[full transfer characterization]\label{thm:main}
Let $k\ge0$ and $n\ge1$.

\smallskip
\noindent\textbf{(A) The weighted composition operator.}
The operator $W=M_\pi C_\varphi$ is $k$-quasi $n$-power posinormal with
constant $\lambda$ if and only if the following corresponding condition holds.

If $k\ge n$, with $r=k-n$, then
\[
 \boxed{\quad
 \Perr_r\bigl(q_n^2|\pi_r|^2\bigr)
 \le\lambda^2q_{k+1}\quad\text{a.e.}
 \quad} \tag{3.2}
\]
If $k<n$, with $r=n-k$, then
\[
 \boxed{\quad
 \Perr_r\left(
   \frac{|\pi_r|^2q_k^2}{q_{k+1}}
 \right)\le\lambda^2\quad\text{a.e.}
 \quad} \tag{3.3}
\]
with the convention (2.4).  Both conditions give the optimal squared
constant by taking the essential supremum of the left/right ratio in (3.2),
or of the left side in (3.3).

\smallskip
\noindent\textbf{(B) The adjoint.}
Let $S_k=\supp q_k$ modulo null sets.  The adjoint $W^*$ is $k$-quasi
$n$-power posinormal with constant $\lambda$ if and only if
\[
 \boxed{\quad
 \int_Xq_n|g|^2d\mu
 \le\lambda^2\|\Perr_1(\overline\pi g)\|_2^2
 \quad\text{for every }g\in L^2(S_k).
 \quad} \tag{3.4}
\]
Equivalently, on $L^2(\mu)$,
\[
 M_{\1_{S_k}q_n}
 \le\lambda^2 M_{\1_{S_k}},W W^*,M_{\1_{S_k}},
 \qquad
 WW^*=M_\pi C_\varphi\Perr_1M_{\overline\pi}. \tag{3.5}
\]
Thus (3.4) is an exact one-step transfer criterion involving only $q_n$, the
support of $q_k$, and the original weighted composition operator.
\end{theorem}

\subsection*{Proof intuition}
All higher powers have a simple transfer normal form:
$W^j=M_{\pi_j}C^j$ and $W^{*j}W^j=M_{q_j}$.  In the regime $k\ge n$, the
middle $W^{*n}W^n$ collapses to a multiplier, leaving another multiplier after
one transfer.  In the regime $k<n$, the uncancelled adjoint power leaves the
map $\Perr_{n-k}M_b$; Lemma~\ref{lem:transfer} computes its exact norm relative
to the denominator $q_{k+1}$.  For $W^*$, the only vectors that matter are in
$\overline{\Ran W^{*k}}=L^2(S_k)$, reducing every power condition to a
one-step lower transfer bound on that support.

\begin{proof}[Proof of Theorem~\ref{thm:main}]
From the definitions and (2.2),
\[
 W^j=M_{\pi_j}C^j,
 \qquad
 W^{*j}W^j=C^{*j}M_{|\pi_j|^2}C^j=M_{q_j}. \tag{3.6}
\]
Condition (3.1) for $W$ is equivalent to
\[
 \|W^{*n}W^kf\|_2^2
 \le\lambda^2\|W^{k+1}f\|_2^2
 =\lambda^2\int_Xq_{k+1}|f|^2d\mu. \tag{3.7}
\]

Suppose first that $k\ge n$ and set $r=k-n$.  Using (3.6),
\[
 W^{*n}W^k
 =W^{*n}W^nW^r
 =M_{q_n\pi_r}C^r.
\]
Hence
\[
 (W^{*n}W^k)^*(W^{*n}W^k)
 =C^{*r}M_{q_n^2|\pi_r|^2}C^r
 =M_{\Perr_r(q_n^2|\pi_r|^2)}. \tag{3.8}
\]
Comparing this multiplier with $M_{q_{k+1}}$ proves (3.2).

Now suppose $k<n$ and put $r=n-k$.  Again by (3.6),
\[
 W^{*n}W^k
 =W^{*r}(W^{*k}W^k)
 =C^{*r}M_{\overline{\pi_r}q_k}
 =\Perr_rM_{\overline{\pi_r}q_k}. \tag{3.9}
\]
Apply Lemma~\ref{lem:transfer} to (3.7) with
$b=\overline{\pi_r}q_k$ and $q=q_{k+1}$.  This is exactly (3.3).

For part (B), apply (3.1) to $W^*$.  With $g=W^{*k}f$, it becomes
\[
 \|W^ng\|_2^2\le\lambda^2\|W^*g\|_2^2. \tag{3.10}
\]
Equation (3.6) gives $\|W^ng\|_2^2=\int q_n|g|^2d\mu$ and
$W^*g=\Perr_1(\overline\pi g)$.  Moreover,
\[
 \overline{\Ran W^{*k}}=(\ker W^k)^\perp=L^2(S_k), \tag{3.11}
\]
because $\|W^kf\|_2^2=\int q_k|f|^2d\mu$.  Both sides of (3.10) are
continuous, so (3.10) on $\Ran W^{*k}$ is equivalent to (3.4) on its closure.
Taking quadratic forms gives (3.5).
\end{proof}

\begin{corollary}[ordinary composition operators]\label{cor:composition}
Let
\[
 h_j=\frac{d(\mu\circ\varphi^{-j})}{d\mu}=\Perr_j(1).
\]
Then $C_\varphi$ is $k$-quasi $n$-power posinormal precisely when
\[
 \begin{cases}
 \Perr_{k-n}(h_n^2)\le\lambda^2h_{k+1},&k\ge n,\\[3pt]
 \displaystyle
 \Perr_{n-k}\!\left(\frac{h_k^2}{h_{k+1}}\right)\le\lambda^2,&k<n.
 \end{cases} \tag{3.12}
\]
Its adjoint is $k$-quasi $n$-power posinormal precisely when
\[
 \int_Xh_n|g|^2d\mu\le\lambda^2\|\Perr_1g\|_2^2
 \quad(g\in L^2(\supp h_k)). \tag{3.13}
\]
\end{corollary}

\begin{proof}
Take $\pi=1$, so that $\pi_j=1$ and $q_j=h_j$, in
Theorem~\ref{thm:main}.
\end{proof}

When $\varphi^{-j}(\Sigma)=\Sigma$ and inverse representatives are available,
$\Perr_j(a)=h_j(a\circ\varphi^{-j})$.  Thus the transfer criteria collapse to
the familiar pointwise expressions in Radon--Nikodym derivatives and inverse
iterates.  The formulas above remain valid when no inverse exists and when the
conditional expectation onto $\varphi^{-j}(\Sigma)$ is nontrivial.

\section{A genuinely noninvertible example}

\begin{example}
Let $X=\{0,1,2\}$ with counting measure and
\[
 \varphi(0)=0,\qquad \varphi(1)=0,\qquad \varphi(2)=1.
\]
Then $\varphi^{-1}(\Sigma)\ne\Sigma$ and
\[
 h_1=(2,1,0),\qquad h_j=(3,0,0)\quad(j\ge2).
\]
For $k=2,n=1$, (3.12) gives
\[
 \Perr_1(h_1^2)=(5,0,0)
 \le\lambda^2(3,0,0)=\lambda^2h_3.
\]
Consequently $C_\varphi$ is $2$-quasi $1$-power posinormal, with optimal
$\lambda^2=5/3$.  On the other hand, for $k=n=1$ the inequality
$h_1^2\le\lambda^2h_2$ fails at the point $1$, so $C_\varphi$ is not
$1$-quasi $1$-power posinormal.  This example lies strictly outside the
source's full-preimage sigma-algebra hypothesis.

For the adjoint and $k=2,n=1$, (3.13) is tested only on functions supported at
$0$ and its optimal constant is $\lambda^2=2$.  For $k=1$, the vector
$g=(1,-1,0)$ is supported on $\supp h_1$, has $\Perr_1g=0$, and has positive
$h_1$-norm; hence the corresponding property fails.  This also illustrates
why the support compression in (3.4) is essential.
\end{example}

\section{Verification, novelty, and limitations}

The proof is analytic and does not depend on computation.  As an independent
check, the included verifier constructs weighted $\ell^2$ matrices for 46
finite systems (43 noninjective), uses real zero/nonzero moduli and complex
phases, and tests all $(k,n)$ with $0\le k\le4$ and $1\le n\le4$.  Across 920
cases it checks (3.6), both factorizations (3.8)--(3.9), the multiplier
formulas, and the optimal constants from (3.2)--(3.3) against direct singular
values.  The recorded verdict is PASS; see
\texttt{VERIFIER\_REPORT.md}.

A bounded search through the run indexes and arXiv on 11 August 2026, using
the exact future-work sentence, title, arXiv identifier, author combinations,
and the phrases ``$k$-quasi $n$-power posinormal'' and ``transfer operator,''
found the source, its published version, and a companion preprint, but no
paper removing this hypothesis or stating Theorem~\ref{thm:main}.  This is a
novelty check, not a guarantee of priority.

The result addresses the first future-work direction for the composition and
weighted-composition results on $L^2$.  It does not address the source's other
directions concerning $L^p$ for $p\ne2$, noncommutative spaces, Hardy/Bergman
spaces, Lambert operators, directed-tree shifts, or applications.  The
boundedness of $C_\varphi$ and the nonsingularity of $\varphi$ are retained;
the removed condition is exactly $\varphi^{-j}(\Sigma)=\Sigma$.
