\section{The question}
For a measurable exponent $p:[0,1]\to[1,\infty)$, write
\[
 p^-=\operatorname*{ess\,inf}p(x),\qquad
 p^+=\operatorname*{ess\,sup}p(x),
\]
and let $D_+([0,1])$ consist of those $p$ with
$1\le p^-\le p(x)\le p^+<\infty$ almost everywhere.  The source uses
$AAS^{p(x)}(\R:X)$ for Stepanov $p(x)$-almost automorphic functions and
$AAAS^{p(x)}([0,\infty):X)$ for their asymptotic version.

\small Figure 1: The exact problem in the archived source PDF, printed page 9.

\paragraph{Transcription.}
Does $p\in D_+([0,1])$ imply
\[
 L^\infty(\R:X)\cap AAS^{p(x)}(\R:X)
 =L^\infty(\R:X)\cap AAS^1(\R:X)                         \tag{Q1}
\]
and
\[
 L^\infty([0,\infty):X)\cap AAAS^{p(x)}([0,\infty):X)
 =L^\infty([0,\infty):X)\cap AAAS^1([0,\infty):X)?       \tag{Q2}
\]
The answer to both questions is yes.

\section{Proof Intuition}
The sole analytic mechanism is bounded-convergence interpolation.  On a unit
interval, if $u_j\to0$ in $L^1$, all $u_j$ have one common bound, and
$p^+<\infty$, then $u_j\to0$ in $L^{p(x)}$.  The variable-exponent modular is
bounded by a fixed multiple of $\|u_j\|_1$.

For (Q1), the local $L^1$ limit profile of translates of a bounded function
inherits the same essential bound.  Both defining limits upgrade from $L^1$
to $L^{p(x)}$.

For (Q2), a witnessing decomposition $f=g+q$ need not initially present $g$
and $q$ as bounded.  Translate it along integers tending to infinity.  The
remainder disappears locally in $L^1$, so the forward almost-automorphic
limit of $g$ is also a limit of translates of bounded $f$.  The reverse limit
then forces $g$ itself to be bounded.  Thus $q=f-g$ is bounded too.

\section{Two lemmas}
The Luxemburg norm on $L^{p(x)}([0,1]:X)$ is
\[
 \|u\|_{p(x)}
 =\inf\left\{\lambda>0:
 \int_0^1\left(\frac{\|u(s)\|}{\lambda}\right)^{p(s)}ds\le1
 \right\}.
\]

\begin{lemma}[bounded $L^1$ convergence upgrades]\label{lem:upgrade}
Let $p\in D_+([0,1])$.  Suppose $u_j:[0,1]\to X$ satisfy
$\|u_j(s)\|\le B$ almost everywhere and $\|u_j\|_1\to0$.
Then $\|u_j\|_{p(x)}\to0$.
\end{lemma}

\begin{proof}
Fix $\varepsilon>0$ and put $v_j=\|u_j\|/\varepsilon$.  Since
$1\le p(s)\le p^+$ and $0\le v_j\le B/\varepsilon$,
\[
 v_j^{p(s)}
 \le v_j+v_j^{p^+}
 \le\left[1+\left(\frac B\varepsilon\right)^{p^+-1}\right]v_j.
\]
The modular of $u_j/\varepsilon$ therefore tends to zero.  It is at most one
for all large $j$, so $\|u_j\|_{p(x)}\le\varepsilon$.
\end{proof}

\begin{lemma}[bounded asymptotic data have bounded recurrent part]
\label{lem:parts}
Suppose $f\in L^\infty([0,\infty):X)$ and
\[
 f=g+q,\qquad
 g\in AAS^1(\R:X),\qquad
 \|\widehat q(t)\|_{L^1([0,1]:X)}\longrightarrow0,          \tag{1}
\]
where $\widehat q(t)(s)=q(t+s)$.  Then
\[
 \|g\|_\infty\le\|f\|_\infty,\qquad
 \|q\|_\infty\le2\|f\|_\infty.                              \tag{2}
\]
\end{lemma}

\begin{proof}
Let $M=\|f\|_\infty$.  Apply Stepanov $1$-almost automorphy of $g$ to the
integer sequence tending to infinity.  Passing to a subsequence
$a_j\to\infty$, there is a profile $h$ such that, for every $t\in\R$,
\[
 \|g(a_j+t+\cdot)-h(t+\cdot)\|_1\to0,\qquad
 \|h(t-a_j+\cdot)-g(t+\cdot)\|_1\to0.                       \tag{3}
\]
For each integer $m$, (1) and $f=g+q$ show that
$f(a_j+m+\cdot)\to h(m+\cdot)$ in $L^1([0,1]:X)$.  Every
function on the left is bounded by $M$, so its $L^1$ limit is bounded by $M$
almost everywhere.  Taking all integer $m$ proves $\|h\|_\infty\le M$.
The reverse limit in (3), again on all integer unit intervals, gives
$\|g\|_\infty\le M$.  Finally $q=f-g$ on the half-line, proving (2).
\end{proof}

\section{The affirmative answer}
\begin{theorem}\label{thm:main}
For every complex Banach space $X$ and every $p\in D_+([0,1])$, both
equalities \emph{(Q1)} and \emph{(Q2)} hold.
\end{theorem}

\begin{proof}
The source proves the inclusions from the $p(x)$ classes into the exponent-one
classes~\cite[Theorem 3.7(i)]{source}; they also follow from the
finite-measure embedding $L^{p(x)}([0,1]:X)\hookrightarrow L^1([0,1]:X)$.

Let $f\in L^\infty(\R:X)\cap AAS^1(\R:X)$ and set $M=\|f\|_\infty$.
Given any real sequence, choose a subsequence $a_j$ and a profile $g$ which
realize the two local $L^1$ limits defining $AAS^1$.  On each integer unit
interval, $g$ is an $L^1$ limit of translates of $f$, hence
$\|g\|_\infty\le M$.  For each fixed $t$, both differences
\[
 f(a_j+t+\cdot)-g(t+\cdot),\qquad
 g(t-a_j+\cdot)-f(t+\cdot)
\]
are bounded by $2M$ and tend to zero in $L^1([0,1]:X)$.
Lemma~\ref{lem:upgrade} upgrades both limits to $L^{p(x)}$.
Bounded $g$ is Stepanov $p(x)$-bounded, so
$f\in AAS^{p(x)}(\R:X)$.  This proves (Q1).

Now let
$f\in L^\infty([0,\infty):X)\cap AAAS^1([0,\infty):X)$ and choose a
decomposition (1).  Lemma~\ref{lem:parts} makes both $g$ and $q$ bounded.
The just-proved result gives $g\in AAS^{p(x)}(\R:X)$.  Since $q$ is bounded
and $\|\widehat q(t)\|_1\to0$, Lemma~\ref{lem:upgrade} gives
\[
 \|\widehat q(t)\|_{L^{p(x)}([0,1]:X)}\longrightarrow0.
\]
Boundedness also makes $q$ Stepanov $p(x)$-bounded.  The same decomposition
therefore witnesses
\[
 f\in AAAS^{p(x)}([0,\infty):X),
\]
which proves (Q2).
\end{proof}

\section{Verification, limitations, and novelty}
The script \texttt{code/verify\_modular\_upgrade.py} checks the modular
inequality in Lemma~\ref{lem:upgrade} on 1,200 randomized finite grids, with
variable exponents between $1$ and $9$.  Every check passes.  This is a
normalization check only; the proof is analytic and exact.

The finite upper exponent $p^+<\infty$ is essential to this proof.  The source
notes different behavior when $p(x)=\infty$ on a nontrivial set.  All
equalities are understood modulo almost-everywhere equality.

\paragraph{Novelty check.}
