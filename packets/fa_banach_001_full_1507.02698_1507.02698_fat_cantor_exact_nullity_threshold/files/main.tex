\begin{abstract}
Hewett and Moiola asked whether a positive-measure set with empty interior can
have Sobolev nullity threshold strictly between $0$ and $n/p$, and asked for
the threshold of their generalized Smith--Volterra--Cantor sets.  We prove
that, for every $n\geq1$, $1<p<\infty$, $0<\alpha<1/2$, and
$0<\beta<1-2\alpha$,
\[
 s_{G_{\alpha,\beta}^{(n)}}(p)
 =\frac1p\left(1+\frac{\log2}{\log\alpha}\right).
\]
The set is null at the threshold.  The proof localizes the source paper's
Swiss-cheese non-nullity criterion, obtains the missing upper bound from one
deleted gap at every level, and passes to Cartesian powers by directional
Bessel-potential slicing.
\end{abstract}

\section{The questions and statement}

For $E\subset\R^n$, let $H_E^{s,p}$ be the distributions in
$H^{s,p}(\R^n)$ supported by $E$.  A set is $(s,p)$-null when
$H_F^{s,p}=\{0\}$ for every closed $F\subset E$.  For a set with empty
interior, its nullity threshold $s_E(p)$ is characterized by nullity above
the threshold and non-nullity below it \cite{HM}.

The source questions are reproduced on the next page.  In the notation of
\cite[Example 4.7]{HM}, start with $l_0=1$ and put
\begin{equation}\label{eq:length-recursion}
 l_{j+1}=\frac12(l_j-\beta\alpha^j),\qquad j\geq0.
\end{equation}
At stage $j+1$, remove from the middle of every one of the $2^j$ basic
intervals a gap of length $\beta\alpha^j$.  The remaining compact set is
$G_{\alpha,\beta}\subset[0,1]$, and
$G_{\alpha,\beta}^{(n)}$ denotes its $n$-fold Cartesian power.  Its measure is
positive because
\begin{equation}\label{eq:length-formula}
 2^j l_j=1-\beta\frac{1-(2\alpha)^j}{1-2\alpha}
 \longrightarrow L:=1-\frac{\beta}{1-2\alpha}>0.
\end{equation}

\begin{theorem}[Exact threshold]\label{thm:main}
Let $n\geq1$, $1<p<\infty$, $0<\alpha<1/2$, and
$0<\beta<1-2\alpha$.  Set
\begin{equation}\label{eq:s0}
 \sigma_{\alpha,p}:=
 \frac1p\left(1+\frac{\log2}{\log\alpha}\right).
\end{equation}
Then
\[
 G_{\alpha,\beta}^{(n)}\text{ is $(s,p)$-null}
 \quad\Longleftrightarrow\quad s\geq\sigma_{\alpha,p}.
\]
In particular,
$s_{G_{\alpha,\beta}^{(n)}}(p)=\sigma_{\alpha,p}$, and the set is null at
its threshold.
\end{theorem}

Since $0<\alpha<1/2$,
$0<\sigma_{\alpha,p}<1/p\leq n/p$.  Theorem \ref{thm:main} therefore also
answers the source's existence question affirmatively in every dimension.

\section{Three tools}

We use two results quoted and proved in the source paper.  First, if
$0<s<1/p$, the Bessel capacity of an interval of radius $r\leq1$ satisfies
\begin{equation}\label{eq:ball-capacity}
 A_{s,p}r^{1-sp}\leq \bcap_{s,p}(B_r(x))
 \leq B_{s,p}r^{1-sp}
\end{equation}
with positive constants depending only on $s,p$; see
\cite[Proposition 3.16]{HM}.  Second, for a closed empty-interior
$F\subset\R$, Theorem 2.19 of \cite{HM} gives
\begin{equation}\label{eq:density-criterion}
 F\text{ is $(s,p)$-null}
 \quad\Longleftrightarrow\quad
 \limsup_{\delta\downarrow0}
 \frac{\bcap_{s,p}(B_\delta(x)\setminus F)}{\delta}>0
 \quad\text{for a.e. }x.
\end{equation}

We also use the following sufficient condition, which is the one-dimensional
case of \cite[Theorem 4.6]{HM}.  If
\[
 K=\overline\Omega\setminus\bigcup_i B_{r_i}(x_i)
\]
is a compact Swiss-cheese set with empty interior, $B_r(x)\subset\Omega$,
and $0<s<1/p$, then
\begin{equation}\label{eq:swiss}
 \sum_i r_i^{1-sp}<\frac{A_{s,p}}{B_{s,p}}r^{1-sp}
 \quad\Longrightarrow\quad K\text{ is not $(s,p)$-null}.
\end{equation}

The final tool passes one-dimensional nullity to products.

\begin{lemma}[Directional slicing]\label{lem:slicing}
Let $s\geq0$, $1<p<\infty$, and write
$\R^n=\R\times\R^{n-1}$.  Then
\begin{equation}\label{eq:slicing}
 H^{s,p}(\R^n)\hookrightarrow
 L^p\bigl(\R^{n-1};H^{s,p}(\R)\bigr).
\end{equation}
Consequently, if $E\subset\R$ is closed and $(s,p)$-null, then
$E^n$ is $(s,p)$-null.
\end{lemma}

\begin{proof}
Let $J_s$ be the Bessel lifting on $\R^n$ and $J_s^{(1)}$ the lifting in
the first variable only.  Applied to $J_su$, the operator $J_s^{(1)}J_s^{-1}$
has multiplier
\[
 m_s(\xi,\eta)=
 \left(\frac{1+|\xi|^2}{1+|\xi|^2+|\eta|^2}\right)^{s/2}.
\]
On each coordinate orthant, direct differentiation shows that
\[
 \sup_{(\xi,\eta)}
 \left|\prod_{k:\gamma_k=1}\zeta_k\,
 \partial^\gamma m_s(\zeta)\right|<\infty,
 \qquad \gamma_k\in\{0,1\}.
\]
The Marcinkiewicz multiplier theorem therefore gives
$\|J_s^{(1)}u\|_p\lesssim\|J_su\|_p$.  Fubini is exactly
\eqref{eq:slicing}.

If $u\in H^{s,p}(\R^n)$ is supported by $E^n$, then for almost every
$y\in\R^{n-1}$ its slice $u(\,cdot\,,y)$ belongs to $H^{s,p}(\R)$ and
vanishes almost everywhere off $E$.  Its distributional support is therefore
contained in $E$, so it is zero.  Hence $u=0$.
\end{proof}

\section{The exact one-dimensional threshold}

Put $G=G_{\alpha,\beta}$ and $s_0=\sigma_{\alpha,p}$.  The critical
arithmetic is
\begin{equation}\label{eq:critical}
 a_0:=1-ps_0=-\frac{\log2}{\log\alpha}\in(0,1),
 \qquad \alpha^{a_0}=\frac12.
\end{equation}

\subsection*{Nullity at the threshold}

Fix $x\in G$.  At level $j-1$, choose a basic interval $I_{j-1}(x)$ of
length $l_{j-1}$ containing $x$.  The central gap removed from it at level
$j$ has length
\[
 g_j=\beta\alpha^{j-1}.
\]
Both $x$ and this gap lie in $I_{j-1}(x)$, so
$B_{2l_{j-1}}(x)\setminus G$ contains an interval of radius $g_j/2$.
Since $l_{j-1}\leq2^{-(j-1)}$, monotonicity and
\eqref{eq:ball-capacity} give
\begin{align*}
 \frac{\bcap_{s_0,p}(B_{2l_{j-1}}(x)\setminus G)}{2l_{j-1}}
 &\geq c\,\frac{g_j^{a_0}}{l_{j-1}}\\
 &\geq c_{\alpha,\beta,p}
       \bigl(2\alpha^{a_0}\bigr)^{j-1}
 =c_{\alpha,\beta,p}>0.
\end{align*}
The radii $2l_{j-1}$ tend to zero.  If $x\notin G$, then $G^c$ is open;
for all sufficiently small $\delta$, the ball $B_\delta(x)$ is contained in
$G^c$, and \eqref{eq:ball-capacity} makes the quotient in
\eqref{eq:density-criterion} tend to infinity.  Thus
\eqref{eq:density-criterion} holds everywhere, and $G$ is
$(s_0,p)$-null.  Standard Sobolev embeddings imply nullity for every
$s\geq s_0$.

\subsection*{Non-nullity below the threshold}

It is enough to consider $0<s<s_0$; positive measure handles $s\leq0$.
Write
\[
 a:=1-sp>0,\qquad q:=2\alpha^a.
\]
By \eqref{eq:critical}, $s<s_0$ is equivalent to $q<1$.

Choose one level-$J$ basic interval $I_J$ and put $K_J=G\cap I_J$.
Inside $I_J$, the global level-$k$ gaps, $k>J$, have radius $g_k/2$ and
multiplicity $2^{k-J-1}$.  Hence
\begin{align}\label{eq:tail}
 S_J&:=\sum_{k>J}2^{k-J-1}\left(\frac{g_k}{2}\right)^a
 \leq C_{\alpha,\beta,s,p}\,2^{-J}q^J.
\end{align}
By \eqref{eq:length-formula}, $l_J\geq L2^{-J}$.  The interior of $I_J$
contains a ball of radius $r_J=l_J/4$, and therefore
\begin{equation}\label{eq:ratio}
 \frac{S_J}{r_J^a}
 \leq C_{\alpha,\beta,s,p}(2\alpha)^{aJ}\longrightarrow0.
\end{equation}
For all sufficiently large $J$, \eqref{eq:ratio} is stronger than
\eqref{eq:swiss}.  Thus $K_J$, and hence $G$, is not $(s,p)$-null.
This proves the one-dimensional case of Theorem \ref{thm:main}.

\section{Cartesian powers and consequences}

At and above $s_0$, Lemma \ref{lem:slicing} transfers the nullity of $G$
to $G^n$.  Conversely, let $s<s_0$ and choose $t$ with
$\max\{s,0\}<t<s_0$.  The one-dimensional result gives a nonzero
$u\in H^{t,p}(\R)$ supported by $G$.  The tensor-product estimate recorded
in \cite[Proposition A.1]{HM} gives
$u^{\otimes n}\in H^{s,p}(\R^n)$; it is nonzero and supported by $G^n$.
For $s\leq0$, the positive measure of $G^n$ also gives non-nullity directly.
This completes the proof of Theorem \ref{thm:main}.

\begin{corollary}[Intermediate thresholds in every dimension]
For every $n\geq1$ and $1<p<\infty$, there is a compact positive-measure
empty-interior set $E\subset\R^n$ satisfying
$0<s_E(p)<n/p$.  Indeed any $G_{\alpha,\beta}^{(n)}$ does so.
\end{corollary}

\begin{remark}
For the classical Smith--Volterra--Cantor choice
$\alpha=\beta=1/4$, Theorem \ref{thm:main} gives
$s_{G_{1/4,1/4}^{(n)}}(p)=1/(2p)$, and threshold nullity.  The parameter
$\beta$ changes the measure and constants but not the exponent.
\end{remark}

\section{Novelty and review note}

A bounded search on 11 August 2026 covered the run registry, the exact paper
title, the phrase ``nullity threshold,'' and the two quoted questions.  It
found the source and bibliographic copies, but no later primary paper stating
Theorem \ref{thm:main}.  The proof uses the source's own capacity tools in a
localized way, so novelty confidence is moderate rather than definitive.
Specialist review should focus first on Lemma \ref{lem:slicing} and on the
critical use of \eqref{eq:density-criterion}.
