\begin{abstract}
Kosti\'c, Pilipovi\'c, and Velinov ask for conditions under which the unique
full-line almost-periodic extension of a positive $C$-distribution cosine
orbit is even.  We prove that no extra condition is required: their existing
hypotheses already force $\Cb(-t)x=\Cb(t)x$.  The proof combines recurrent
positive almost periods with the cosine identity and closedness of the
associated first-order $C$-distribution-semigroup operator.  Passing through
the associated $2\times2$ operator avoids any unsupported assumption that an
individual unbounded cosine component is closed.
\end{abstract}

\section{The question and the answer}

Let $E$ be a complex Banach space, let $C\in L(E)$ be injective, and let
$\mathbf G$ be a $C$-distribution cosine function generated by a closed
operator $A$, with the notation of Kosti\'c--Pilipovi\'c--Velinov
\cite{KPV}.  Its solution space for the second-order problem is denoted by
$Z_2(A)$.  For $x\in Z_2(A)$ and $t\geq0$, write
\[
  C_x(t):=\Gop(\delta_t)x.
\]
The source recalls the invariance and cosine identities
\begin{align}
 \Gop(\delta_t)Z_2(A)&\subseteq Z_2(A), \label{eq:invariance}\\
 2\Gop(\delta_s)\Gop(\delta_t)x
   &=\Gop(\delta_{s+t})x+\Gop(\delta_{|s-t|})x
       \qquad(s,t\geq0). \label{eq:cosine}
\end{align}

Suppose that $\Eap\subseteq Z_2(A)$ is a linear subspace and that
$C_x\in AP([0,\infty);E)$ for every $x\in\Eap$.  If
\[
 F:AP([0,\infty);E)\longrightarrow AP(\R;E)
\]
is the source's unique isometric extension map, put
$\Cb_x:=F(C_x)$.  The question in item 3 on page 10 of \cite{KPV} asks what
conditions ensure $\Cb_x(-t)=C_x(t)$.

\begin{theorem}\label{thm:main}
Under precisely the hypotheses above, with no additional invariance,
closedness, or boundedness assumption on $\Eap$, one has
\[
  \Cb_x(-t)=\Cb_x(t)=\Gop(\delta_t)x
  \qquad(t\geq0,\ x\in\Eap).
\]
Thus the canonical almost-periodic extension of every such orbit is even.
\end{theorem}

\section{Two elementary inputs}

We isolate the two points that make the limit argument rigorous.

\begin{lemma}[half-line periods remain full-line periods]\label{lem:period}
Let $f\in AP([0,\infty);E)$ and $g=Ff$.  For every $\tau\geq0$,
\[
 \sup_{r\in\R}\|g(r+\tau)-g(r)\|
 =\sup_{r\geq0}\|f(r+\tau)-f(r)\|.
\]
\end{lemma}

\begin{proof}
The full-line almost-periodic function $r\mapsto g(r+\tau)-g(r)$ restricts on
$[0,\infty)$ to $r\mapsto f(r+\tau)-f(r)$.  By uniqueness of $F$, it is the
$F$-extension of that restriction.  The equality is exactly the isometry of
$F$.
\end{proof}

For the second input, recall the source's first-order reduction
\[
 \Acal=\begin{pmatrix}0&I\\ A&0\end{pmatrix},\qquad
 \Ccal=\begin{pmatrix}C&0\\0&C\end{pmatrix}.
\]
The associated $\Ccal$-distribution semigroup on $E\oplus E$ will be denoted
by $\GG$.  For every compactly supported distribution $T$, its operator
$\GG(T)$ is a closed linear operator by the general construction recalled in
\cite[Section~1]{KPV}.

\begin{lemma}[orbit of the first-order reduction]\label{lem:block}
For $y\in Z_2(A)$ and $t\geq0$,
\[
 \GG(\delta_t)\binom{0}{y}
 =\binom{\displaystyle\int_0^t\Gop(\delta_s)y\,ds}
             {\Gop(\delta_t)y}.                                      \tag{2.1}
\]
\end{lemma}

\begin{proof}
Put $u(s)=\Gop(\delta_s)y$ and
$U(s)=(\int_0^s u(r)\,dr,u(s))^T$.  The second-order mild identity gives
\[
 \int_0^t(t-r)u(r)\,dr\in D(A),\qquad
 A\int_0^t(t-r)u(r)\,dr=u(t)-y.
\]
Consequently
\[
 \Acal\int_0^tU(s)\,ds
 =\binom{\int_0^tu(s)\,ds}{u(t)-y}
 =U(t)-U(0).
\]
Thus $U$ is the first-order mild solution with initial value $(0,y)^T$.
The source characterization of the solution space of a distribution
semigroup identifies it with the orbit of $\GG(\delta_t)$, proving (2.1).
\end{proof}

\section{Proof of Theorem~\ref{thm:main}}

Fix $x\in\Eap$ and abbreviate
\[
 f(s)=\Gop(\delta_s)x\quad(s\geq0),\qquad g=Ff.
\]
Fix $t>0$.  For each $n$, relative density of the positive
$1/n$-periods of $f$ lets us choose such a period $\tau_n>t+n$.  By
Lemma~\ref{lem:period},
\begin{equation}
 \sup_{r\in\R}\|g(r+\tau_n)-g(r)\|\leq\frac1n.       \label{eq:period}
\end{equation}
In particular, with $y_n=f(\tau_n)$, we have $y_n\to g(0)=x$.

By \eqref{eq:invariance}, $y_n\in Z_2(A)$.  For $0\leq s\leq t$, the
choice $\tau_n>t$ and \eqref{eq:cosine} give
\begin{align*}
 \Gop(\delta_s)y_n
 &=\frac12\bigl(f(\tau_n+s)+f(\tau_n-s)\bigr)\\
 &=\frac12\bigl(g(\tau_n+s)+g(\tau_n-s)\bigr).
\end{align*}
The full-line estimate \eqref{eq:period} therefore yields, uniformly for
$0\leq s\leq t$,
\begin{equation}
 \Gop(\delta_s)y_n\longrightarrow
 p(s):=\frac12\bigl(g(s)+g(-s)\bigr).                 \label{eq:uniform}
\end{equation}

Apply Lemma~\ref{lem:block}.  Uniform convergence in \eqref{eq:uniform}
shows that
\[
 \GG(\delta_t)\binom{0}{y_n}
 =\binom{\int_0^t\Gop(\delta_s)y_n\,ds}
             {\Gop(\delta_t)y_n}
 \longrightarrow
 \binom{\int_0^tp(s)\,ds}{p(t)}.
\]
Since $(0,y_n)^T\to(0,x)^T$ and the operator $\GG(\delta_t)$ is closed,
the limit is $\GG(\delta_t)(0,x)^T$.  Lemma~\ref{lem:block}, now for $x$,
identifies its second coordinate as
\[
 p(t)=\Gop(\delta_t)x=f(t)=g(t).
\]
By the definition of $p(t)$ this is
$\tfrac12(g(t)+g(-t))=g(t)$, hence $g(-t)=g(t)$.  The case $t=0$ is
immediate, completing the proof. \qed

\section{Checks and consequences}

\begin{itemize}[leftmargin=2em]
\item The proof never applies a closed-graph argument directly to the
possibly unbounded component $\Gop(\delta_t)$.  It proves convergence of both
coordinates and uses the explicitly closed associated operator
$\GG(\delta_t)$.
\item No assumption that $\Gop(\delta_t)\Eap\subseteq\Eap$ is needed.  Only
the source's automatic invariance of $Z_2(A)$ is used.
\item The argument is orbitwise.  It does not require $\Eap$ to be closed or
dense and does not require uniform operator bounds.
\item The theorem answers the displayed even-extension question only.  It
does not address the distinct weak-almost-periodicity problem at the end of
the source paper.
\end{itemize}
