\section{Source question}

The source is Narinder S. Claire, \emph{Functional Calculus for Semi-Bounded
Operators} \cite{Claire}, arXiv:\allowbreak math/0212093.  Its standing hypothesis is that
$H$ is closed and densely defined on a Banach space, $\sigma(H)\subset\R$, and
for some $\alpha\geq0$,
\begin{equation}\label{eq:source-resolvent}
 \norm{(z-H)^{-1}}
 \leq c\,|\operatorname{Im}z|^{-1}
 \left(\frac{\angles z}{|\operatorname{Im}z|}\right)^\alpha,
 \qquad z\notin\R.
\end{equation}
For $\beta\in\R$, the symbol class $S^\beta$ consists of the smooth functions
$g$ such that
\begin{equation}\label{eq:symbol}
 |g^{(k)}(x)|\leq C_k\angles{x}^{\beta-k}
 \quad(k=0,1,\ldots).
\end{equation}
The paper builds a multiplicative bounded-operator calculus on
$\A=\bigcup_{\beta<0}S^\beta$ and says in Remark~1.4 (page 2):
\begin{quote}
``The strict inequality $\beta<0$ is of importance and we conjecture that it
cannot be extended to $\beta\leq0$.''
\end{quote}

\begin{figure}[htbp]
  \centering
  \includegraphics[width=0.93\textwidth]{figures/open_conjecture_crop.png}
  \caption{The definition of $\A$ and Remark~1.4 in the source PDF, page 2.}
\end{figure}

Since $S^\beta\subset S^0$ for $\beta<0$, the proposed endpoint is all of
$S^0$.  Constants alone do not settle the conjecture: they can be adjoined by
unitalization.  We therefore use the following strong and natural formulation.

\medskip
\noindent\textbf{Endpoint extension problem.}
Must every operator satisfying \eqref{eq:source-resolvent} admit an algebra
homomorphism
\[
 \Gamma:S^0(\R)\longrightarrow\B(X)
\]
whose restriction to $\A$ is the Davies--Helffer--Sj\"ostrand calculus?

\section{Proof intuition}

Smooth order-zero symbols may oscillate indefinitely on a logarithmic scale.
We encode exactly those oscillations into a conditional Schauder basis.  On
the summing basis $s_n=e_1+\cdots+e_n$ of $c_0$, the diagonal operator
$Hs_n=e^ns_n$ has an exceptionally good resolvent: the coefficients
$(z-e^n)^{-1}$ have total variation at most $\pi/|\operatorname{Im}z|$.
Thus $H$ satisfies the source hypothesis already with $\alpha=0$.

On the other hand, a smooth order-zero function can satisfy
$f(e^n)=(-1)^n$.  If an extension $\Gamma$ existed, compactly supported
spectral localizers and multiplicativity would force
$\Gamma(f)s_n=(-1)^ns_n$.  That is the alternating-sign multiplier on the
summing basis, which is unbounded.  The obstruction therefore concerns every
possible multiplicative extension, not just convergence of the integral
formula used in the source.

\section{The result}

\begin{theorem}[Order-zero endpoint obstruction]\label{thm:main}
There exist a complex Banach space $X$ and a closed densely defined operator
$H$ on $X$ such that
\[
 \sigma(H)=\{e^n:n\geq1\}\subset(0,\infty)
\]
and
\begin{equation}\label{eq:sharp}
 \norm{(z-H)^{-1}}\leq \frac{2\pi}{|\operatorname{Im}z|},
 \qquad z\notin\R.
\end{equation}
Nevertheless, there is no algebra homomorphism
\[
 \Gamma:S^0(\R)\longrightarrow\B(X)
\]
that agrees on $\A=\bigcup_{\beta<0}S^\beta$ with the
Davies--Helffer--Sj\"ostrand functional calculus of $H$.

More explicitly, the failure is witnessed by a real-valued $f\in S^0(\R)$.
No continuity assumption on $\Gamma$ is required.
\end{theorem}

\begin{corollary}\label{cor:halfline}
The analogous universal extension statement from $\A^+$ to $S^0_+$ for the
semibounded calculus in Claire's paper is false as well.
\end{corollary}

\section{Proof}

\subsection{The summing basis and a variation estimate}

Let $X=c_0$ with its usual norm and standard unit vectors $(e_k)$.  Put
\[
 s_n=e_1+\cdots+e_n \qquad(n\geq1).
\]
This is the summing Schauder basis.  Indeed, if $x=(x_k)\in c_0$ and
$a_n=x_n-x_{n+1}$, then
\[
 x=\sum_{n=1}^\infty a_ns_n.
\]
The $N$th basis projection $P_Nx=\sum_{n\leq N}a_ns_n$ has coordinates
\[
 (P_Nx)_k=
 \begin{cases}x_k-x_{N+1},&k\leq N,\\0,&k>N,
 \end{cases}
\]
so $P_Nx\to x$ and
\begin{equation}\label{eq:basisconstant}
 K:=\sup_N\norm{P_N}\leq2.
\end{equation}

We record the elementary multiplier estimate used below.

\begin{lemma}[Bounded-variation multiplier]\label{lem:bv}
Let $(u_n)$ be a Schauder basis with basis constant $K$.  If $c_n\to0$ and
\[
 V(c):=\sum_{n=1}^\infty |c_n-c_{n+1}|<\infty,
\]
then the diagonal rule $M_cu_n=c_nu_n$ extends to a bounded operator and
$\norm{M_c}\leq K V(c)$.
\end{lemma}

\begin{proof}
Writing $P_n$ for the basis projections, Abel summation gives
\[
 M_cx=\sum_{n=1}^\infty(c_n-c_{n+1})P_nx.
\]
The series converges absolutely in operator norm and has norm at most
$KV(c)$.  Its value on every basis vector is $c_nu_n$, which proves the
claim.
\end{proof}

\subsection{A closed multiplier with positive spectrum}

Let $\lambda_n=e^n$ and define the Schauder multiplier
\begin{align*}
 D(H)&=\left\{x=\sum_{n\geq1}a_ns_n:
                 \sum_{n\geq1}\lambda_na_ns_n\text{ converges in }c_0\right\},\\
 Hx&=\sum_{n\geq1}\lambda_na_ns_n.
\end{align*}
The finite linear span of the $s_n$ lies in $D(H)$, hence $H$ is densely
defined.  It is closed: if $x^{(j)}\to x$ and $Hx^{(j)}\to y$, continuity of
the basis coefficient functionals gives
\[
 a_n(y)=\lambda_na_n(x) \qquad(n\geq1).
\]
The Schauder expansion of $y$ therefore shows that the defining series for
$Hx$ converges to $y$.

Fix $z=x+iy\notin\R$ and set $c_n=(z-e^n)^{-1}$.  The chord joining two
successive reciprocal values is no longer than the intervening curve, so
\begin{align}
 \sum_{n=1}^\infty|c_n-c_{n+1}|
 &\leq \sum_{n=1}^\infty
   \int_{e^n}^{e^{n+1}}\frac{dt}{|z-t|^2}\notag\\
 &=\int_e^\infty\frac{dt}{(t-x)^2+y^2}
 \leq \frac{\pi}{|y|}.\label{eq:variation}
\end{align}
Lemma~\ref{lem:bv} and \eqref{eq:basisconstant} give a bounded diagonal
operator $R_z=M_c$ satisfying \eqref{eq:sharp}.  Moreover,
\[
 e^nc_n=zc_n-1.
\]
Consequently $R_zx\in D(H)$ and $(z-H)R_zx=x$.  Conversely, comparison of
basis coefficients gives $R_z(z-H)u=u$ for $u\in D(H)$.  Thus
$R_z=(z-H)^{-1}$.

The same argument identifies the entire spectrum.  For real
$t\notin\{e^n:n\geq1\}$, the sequence $(t-e^n)^{-1}$ tends to zero and has
finite total variation: after finitely many terms it is monotone, and the
remaining finite initial segment causes no problem.  The preceding inverse
argument therefore shows $t\in\rho(H)$.  Each $e^n$ is an eigenvalue with
eigenvector $s_n$.  Hence
\[
 \sigma(H)=\{e^n:n\geq1\}.
\]
This also verifies the source hypothesis with its strongest allowed exponent
$\alpha=0$.

\subsection{One order-zero symbol encodes the bad signs}

Choose a real $\chi\in C^\infty(\R)$ with $\chi=0$ on $(-\infty,1]$ and
$\chi=1$ on $[e,\infty)$.  Define
\begin{equation}\label{eq:f}
 f(t)=
 \begin{cases}
   \chi(t)\cos(\pi\log t),&t>0,\\
   0,&t\leq0.
 \end{cases}
\end{equation}
This is smooth because it vanishes on a neighborhood of the origin.  For
$t\geq e$, repeated differentiation expresses $f^{(k)}(t)$ as
$t^{-k}$ times a fixed linear combination of
$\sin(\pi\log t)$ and $\cos(\pi\log t)$.  On the compact transition interval
$[1,e]$ all derivatives are bounded.  Therefore
\[
 |f^{(k)}(t)|\leq C_k\angles t^{-k},
\]
so $f\in S^0(\R)$.  Its spectral values are
\begin{equation}\label{eq:signvalues}
 f(e^n)=\cos(\pi n)=(-1)^n.
\end{equation}

\subsection{Multiplicativity forces an unbounded operator}

Suppose, toward a contradiction, that $\Gamma:S^0(\R)\to\B(c_0)$ is an
algebra homomorphism extending the existing calculus on $\A$.  For each $n$
choose $\phi_n\in C_c^\infty(\R)$ such that $\phi_n(e^n)=1$ and whose support
contains no other $e^j$.  Both $\phi_n$ and $f\phi_n$ belong to
$C_c^\infty(\R)\subset\A$.

The source calculus has the usual eigenvector identity
\begin{equation}\label{eq:eigenidentity}
 g(H)s_j=g(e^j)s_j\qquad(g\in C_c^\infty(\R)).
\end{equation}
For completeness, applying its almost-analytic integral to $s_j$ replaces
$(z-H)^{-1}s_j$ by $(z-e^j)^{-1}s_j$; the scalar Cauchy--Green identity then
gives \eqref{eq:eigenidentity}.  Agreement and multiplicativity now imply
\begin{align*}
 \Gamma(f)\phi_n(H)
 &=\Gamma(f)\Gamma(\phi_n)
  =\Gamma(f\phi_n)=(f\phi_n)(H).
\end{align*}
Applying this to $s_n$ and using \eqref{eq:eigenidentity} gives
\begin{equation}\label{eq:forced}
 \Gamma(f)s_n=f(e^n)s_n=(-1)^ns_n.
\end{equation}

But the multiplier in \eqref{eq:forced} is not bounded on the summing basis.
Indeed, let
\[
 x_N=\sum_{n=1}^N(-1)^ns_n.
\]
Every coordinate of $x_N$ is an alternating tail sum, so
$\norm{x_N}_\infty=1$.  Equation \eqref{eq:forced} would give
\[
 \Gamma(f)x_N=\sum_{n=1}^Ns_n,
 \qquad
 \norm{\Gamma(f)x_N}_\infty=N,
\]
contradicting $\Gamma(f)\in\B(c_0)$.  This proves
Theorem~\ref{thm:main}.

Finally, restrict $f$ and the compactly supported localizers to
$[0,\infty)$.  Claire's semibounded calculus agrees with the whole-line
calculus after smooth extension and has the same eigenvector identities.
The identical contradiction proves Corollary~\ref{cor:halfline}.

\section{Verification notes and limitations}

The proof is exact and uses no numerical input.  The main review checkpoints
are: (i) the Abel-summation multiplier estimate; (ii) closedness and the
two-sided inverse on the stated multiplier domain; (iii) the logarithmic
derivative bounds in \eqref{eq:f}; and (iv) the localization step forcing
\eqref{eq:forced}.

The result resolves the source conjecture in the universal extension sense
stated above.  It does \emph{not} assert that every individual order-zero
symbol is obstructed, nor that no proper subalgebra of $S^0$ can be adjoined.
In particular, adjoining constants is harmless.  It also does not claim that
Claire supplied a more formal endpoint statement than Remark~1.4; the packet
makes the tested formulation explicit.
