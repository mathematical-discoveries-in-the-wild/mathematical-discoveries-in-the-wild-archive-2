\begin{abstract}
King and Koldan conjecture that the Carlen--Lieb norm and their
Pisier-type noncommutative norm are inequivalent for every
$1<p\le2$ and every $q\ge1$.  The diagonal choice $q=p$ refutes the
statement as written.  In fact, for every matrix size, every
$1\le p\le2$, and every bipartite matrix $Y$,
\[
 \boxed{\|Y\|_{\CL:p,p}=\|Y\|_{\NC:p,p}=\|Y\|_p.}
\]
The source already records the second equality.  We prove the first by
computing the positive-decomposition gauge in the Carlen--Lieb definition:
for Hermitian $X$ it equals $\|X_+\|_p+\|X_-\|_p$.  On the standard
Hermitian dilation of $Y$, both spectral parts have norm $\|Y\|_p$.
This also disproves the source's nonmonotonicity lemma at its stated endpoint
$p=q$.  The off-diagonal conjecture $q\ne p$ is not settled here.
\end{abstract}

\section{The source conjecture}

all matrix sizes and then states the conjecture shown below.

In exact text, the target is:
\begin{quote}
We conjecture that the norms are not equivalent for all
$1<p\le2$ and all $q\ge1$.
\end{quote}
Because ``all $q\ge1$'' includes $q=p$, a single dimension-independent
equivalence on that diagonal is a full counterexample.  We prove the stronger
identity of the two norms there.

\section{Definitions specialized to \texorpdfstring{$q=p$}{q=p}}

For $A\ge0$ in a bipartite matrix algebra, the source defines
\[
 \Psi_{p,q}(A)=
 \left(\tr_1\bigl(\tr_2 A^p\bigr)^{q/p}\right)^{1/q}.
\]
At $q=p$, the two partial traces compose to the full trace, so
\begin{equation}\label{eq:PsiSchatten}
 \Psi_{p,p}(A)=(\tr A^p)^{1/p}=\|A\|_p.
\end{equation}

For Hermitian $X$, introduce the positive-decomposition gauge used in the
Carlen--Lieb construction:
\begin{equation}\label{eq:gauge}
 |||X|||_{p,q}:=
 \inf\{\Psi_{p,q}(A)+\Psi_{p,q}(B):
       A,B\ge0,\ X+A=B\}.
\end{equation}
For arbitrary $Y\in\M_n\otimes\M_m$, let
\[
 H(Y)=\begin{pmatrix}0&Y\\Y^*&0\end{pmatrix}.
\]
The Carlen--Lieb norm is
\begin{equation}\label{eq:CLdef}
 \|Y\|_{\CL:p,q}=\frac12|||H(Y)|||_{p,q},
\end{equation}
where the auxiliary two-dimensional factor is placed in the inside space.
At $p=q$ the splitting is irrelevant by \eqref{eq:PsiSchatten}.

The source's Lemma 1(e) proves directly that the Pisier-type norm satisfies
\begin{equation}\label{eq:NCsource}
 \|Y\|_{\NC:p,p}=\|Y\|_p.
\end{equation}
It remains only to compute \eqref{eq:gauge}.

\section{Exact positive-decomposition gauge}

\paragraph{Proof idea.}
If $X=B-A$ with $A,B\ge0$, then $B\ge X$ and $A\ge-X$.  Eigenvalue
monotonicity forces $B$ to carry at least the positive spectrum of $X$ and
$A$ to carry at least its negative spectrum.  The canonical choice
$B=X_+$, $A=X_-$ attains both lower bounds simultaneously.

\begin{lemma}[one-sided spectral lower bound]\label{lem:one-sided}
Let $X=X^*$ and $P\ge0$.  If $P\ge X$, then for every $1\le p<\infty$,
\[
 \|P\|_p\ge\|X_+\|_p.
\]
\end{lemma}

\begin{proof}
List eigenvalues in decreasing order.  If $X$ has $k$ positive eigenvalues,
the min--max principle and $P\ge X$ give
$\lambda_j(P)\ge\lambda_j(X)>0$ for $1\le j\le k$.  Since $P\ge0$,
\[
 \|P\|_p^p=\sum_j\lambda_j(P)^p
 \ge\sum_{j=1}^k\lambda_j(X)^p=\|X_+\|_p^p.
\]
\end{proof}

\begin{theorem}[exact diagonal gauge]\label{thm:gauge}
For every Hermitian matrix $X$ and $1\le p<\infty$,
\[
 |||X|||_{p,p}=\|X_+\|_p+\|X_-\|_p.
\]
\end{theorem}

\begin{proof}
For any admissible $A,B$ in \eqref{eq:gauge}, $B-X=A\ge0$, hence
$B\ge X$.  Lemma \ref{lem:one-sided} yields
$\|B\|_p\ge\|X_+\|_p$.  Also $A-(-X)=B\ge0$, so applying the lemma to
$-X$ yields $\|A\|_p\ge\|X_-\|_p$.  By \eqref{eq:PsiSchatten}, every
admissible cost is therefore at least the displayed sum.

The spectral decomposition $X=X_+-X_-$ gives the admissible choice
$A=X_-$ and $B=X_+$.  Its cost is exactly the lower bound.
\end{proof}

\section{Full counterexample and a second endpoint correction}

\begin{theorem}[exact equality on the diagonal]\label{thm:main}
For every $1\le p\le2$, all $n,m\ge1$, and every
$Y\in\M_n\otimes\M_m$,
\[
 \|Y\|_{\CL:p,p}=\|Y\|_{\NC:p,p}=\|Y\|_p.
\]
Consequently the all-parameter inequivalence conjecture in arXiv:0904.1710v2
is false for every $1<p\le2$ by choosing $q=p$.
\end{theorem}

\begin{proof}
If $(s_j)$ are the singular values of $Y$, the nonzero spectrum of the
Hermitian dilation $H(Y)$ consists of the pairs $\{s_j,-s_j\}$.  Hence
\[
 \|H(Y)_+\|_p=\|H(Y)_-\|_p
 =\left(\sum_j s_j^p\right)^{1/p}=\|Y\|_p.
\]
Theorem \ref{thm:gauge} and \eqref{eq:CLdef} give
\[
 \|Y\|_{\CL:p,p}
 =\tfrac12\bigl(\|H(Y)_+\|_p+\|H(Y)_-\|_p\bigr)=\|Y\|_p.
\]
Equation \eqref{eq:NCsource} gives the NC identity.  These equalities hold
with constant one independently of $n$ and $m$, exactly contradicting the
source definition of inequivalence.
\end{proof}

\begin{corollary}[endpoint of the source nonmonotonicity lemma]
At $q=p$, the map $Y\mapsto\|Y\|_{\CL:p,p}$ is monotone on positive
semidefinite matrices.  Thus source Lemma 3, which states nonmonotonicity for
all $p\le q$, is also false at the equality endpoint.
\end{corollary}

\begin{proof}
Theorem \ref{thm:main} identifies it with the Schatten $p$-norm, which is
monotone under the Loewner order on positive matrices.
\end{proof}

\section{Verification, novelty boundary, and intended scope}

The computation has three independent checks.  First,
$\Psi_{p,p}$ loses all bipartite dependence and is the Schatten norm.
Second, the min--max lower bound is dimension-free and is attained by the
canonical spectral parts.  Third, the factor $1/2$ in the CL definition is
cancelled exactly by the two equal spectral halves of $H(Y)$.

Targeted searches of the run indexes, the latest arXiv version, and primary
literature searches found the conjecture and the source's NC reduction but no
statement of the CL diagonal identity or correction of the endpoint.  This is
not a comprehensive priority guarantee; specialist review is recommended.

The wording may have intended ``all remaining off-diagonal cases'' or
$q\ne p$: Theorem 4 immediately before the conjecture assumes $p<q$, and
the source also overstates its nonmonotonicity lemma at $p=q$.  This packet
answers the literal quantified conjecture only.  It does not resolve
inequivalence when $q\ne p$.
