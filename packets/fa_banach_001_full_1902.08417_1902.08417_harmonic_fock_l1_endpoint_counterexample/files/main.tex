\begin{abstract}
Vujadinovic conjectures that the harmonic Fock projection $P_\alpha$ is
bounded on $L^p(\R^n,d\mu_\beta)$, for $p\ge1$, if and only if
$p\beta=2\alpha$, and claims this theorem when $n$ is even.  We give an
elementary counterexample at the critical endpoint.  In dimension four,
$P_\alpha$ is not bounded on $L^1(\R^4,d\mu_{2\alpha})$ for any
$\alpha>0$.  The exact four-dimensional kernel turns a column centered at
$Re_1$ into a fixed Gaussian times an oscillatory expression of size $R$.
Its absolute mass is bounded below by $cR-C$.  Since bounded $L^1$ integral
operators have uniformly bounded absolute columns, the result follows.
This disproves Conjecture 1.6(b) and also contradicts the source's claimed
even-dimensional endpoint theorem.
\end{abstract}

\section{Exact source question and definitions}

For $\gamma>0$, let
\[
 d\mu_\gamma(x)=(\pi\gamma)^{-n/2}e^{-|x|^2/\gamma}\dd x.
\]
The harmonic Fock space $\mathcal H^2_\alpha$ consists of the harmonic
functions in $L^2(\R^n,d\mu_\alpha)$, and its orthogonal projection is
\[
 P_\alpha f(x)=\int_{\R^n}H_\alpha(x,y)f(y)\dd\mu_\alpha(y).
\]
The source states:
\begin{quote}
\emph{Conjecture 1.6. Suppose $\alpha>0$, $\beta>0$ and
$0<p<\infty$. Then: (a) $P_\alpha$ is not bounded on
$L^p(\R^n,d\mu_\beta)$ for $0<p<1$; (b) for $p\ge1$, $P_\alpha$ is
bounded on $L^p(\R^n,d\mu_\beta)$ if and only if $p\beta=2\alpha$.}
\end{quote}
The paper's Theorem 1.4(b) claims part (b) for every even $n$.  The exact
source page is reproduced below.
\begin{center}
 \includegraphics[width=0.77\textwidth]{figures/open_problem_crop.png}
\end{center}

We disprove the sufficiency direction already for $n=4$, $p=1$ and
$\beta=2\alpha$.

\section{The exact four-dimensional kernel}

It is enough to take $\alpha=1$.  Indeed, the changes of variables
$x=\sqrt\alpha X$ and $y=\sqrt\alpha Y$ carry $d\mu_\alpha$ to
$d\mu_1$, $d\mu_{2\alpha}$ to $d\mu_2$, and
\[
 H_\alpha(\sqrt\alpha X,\sqrt\alpha Y)=H_1(X,Y).
\]

For completeness, we derive the needed kernel directly from the zonal
series quoted in the source.  Put
\[
 s=|x||y|,\qquad \cos\theta=\frac{\langle x,y\rangle}{|x||y|}.
\]
In dimension four, the normalized zonal harmonic of degree $k$ is
$(k+1)C_k^1(\cos\theta)$ and $(2)_k=(k+1)!$.  Consequently
\begin{align*}
 H_1(x,y)
 &=\sum_{k=0}^\infty C_k^1(\cos\theta)\frac{s^k}{k!}\\
 &=\frac1{\sin\theta}\operatorname{Im}
     \left(e^{i\theta}e^{se^{i\theta}}\right).
\end{align*}
Writing
\[
 A=\langle x,y\rangle=s\cos\theta,
 \qquad
 B=\sqrt{|x|^2|y|^2-\langle x,y\rangle^2}=s\sin\theta,
\]
we obtain the exact formula
\begin{equation}\label{kernel}
 \boxed{\quad
 H_1(x,y)=e^A\left(A\frac{\sin B}{B}+\cos B\right),
 \quad}
\end{equation}
where $\sin B/B$ is interpreted as $1$ at $B=0$.  This also agrees with
Englis's independent closed kernel formula \cite[Corollary 7]{Englis}.

\section{Absolute columns and the counterexample}

Relative to $d\mu_2$, the same projection is
\begin{equation}\label{relative-kernel}
 P_1f(x)=\int_{\R^4}K(x,y)f(y)\dd\mu_2(y),
 \qquad K(x,y)=4e^{-|y|^2/2}H_1(x,y),
\end{equation}
because $d\mu_1(y)=4e^{-|y|^2/2}d\mu_2(y)$.

We first record the elementary endpoint fact that will turn an exact column
calculation into operator unboundedness.

\begin{lemma}[bounded $L^1$ kernels have bounded columns]\label{column-lemma}
Let $\nu$ be a locally finite measure with full support on $\R^d$, and let
$K$ be continuous.  If
\[
 Tf(x)=\int K(x,y)f(y)\dd\nu(y)
\]
extends boundedly on $L^1(\nu)$, then, for every $y$,
\[
 \int |K(x,y)|\dd\nu(x)\le \|T\|_{L^1\to L^1}.
\]
\end{lemma}

\begin{proof}
Let $f_\varepsilon$ be the indicator of $B(y,\varepsilon)$ divided by its
$\nu$-measure.  Then $\|f_\varepsilon\|_1=1$ and, by continuity,
$Tf_\varepsilon(x)\to K(x,y)$ for every $x$.  Fatou's lemma gives
\[
 \int|K(x,y)|\dd\nu(x)
 \le\liminf_{\varepsilon\downarrow0}\|Tf_\varepsilon\|_1
 \le\|T\|.
\]
\end{proof}

\begin{theorem}\label{main}
For every $\alpha>0$, the harmonic Fock projection $P_\alpha$ is not
bounded on $L^1(\R^4,d\mu_{2\alpha})$.
\end{theorem}

\begin{proof}
By scaling, take $\alpha=1$.  For $R>0$, set $y_R=Re_1$ and parameterize
\[
 x=(R+u,v),\qquad u\in\R,\quad v\in\R^3,quad \rho=|v|.
\]
Then $A=R(R+u)$ and $B=R\rho$.  Combining
\eqref{kernel}, \eqref{relative-kernel}, and
$d\mu_2(x)=(2\pi)^{-2}e^{-|x|^2/2}\dd x$ gives the exact absolute column
mass
\begin{align}
 J(R)&:=\int_{\R^4}|K(x,y_R)|\dd\mu_2(x)\notag\\
 &=\frac1{\pi^2}\int_{\R}\int_{\R^3}
 e^{-(u^2+|v|^2)/2}
 \left|(R+u)\frac{\sin(R|v|)}{|v|}+\cos(R|v|)\right|
 \dd v\dd u.\label{exact-column}
\end{align}
The decisive cancellation in the Gaussian exponent is exact:
\[
 -\frac{R^2}{2}-\frac{(R+u)^2+|v|^2}{2}+R(R+u)
 =-\frac{u^2+|v|^2}{2}.
\]

Restrict \eqref{exact-column} to $|u|\le1$ and $1\le\rho\le2$.
For $R\ge2$, reverse triangle inequality gives
\[
 \left|(R+u)\frac{\sin(R\rho)}{\rho}+\cos(R\rho)\right|
 \ge \frac{R-1}{2}|\sin(R\rho)|-1.
\]
Furthermore,
\begin{align*}
 \int_1^2|\sin(R\rho)|\dd\rho
 &=\frac1R\int_R^{2R}|\sin t|\dd t\\
 &\ge \frac{2}{R}\left\lfloor\frac R\pi\right\rfloor
 \ge \frac2\pi-\frac2R.
\end{align*}
On $1\le\rho\le2$, the radial Gaussian density
$e^{-\rho^2/2}\rho^2$ is at least $e^{-2}$.  Passing to polar coordinates
in $v$ therefore yields constants $c,C>0$, independent of $R$, such that
\[
 J(R)\ge cR-C\longrightarrow\infty.
\]
Lemma \ref{column-lemma}, applied to the continuous kernel
\eqref{relative-kernel}, contradicts boundedness on $L^1(d\mu_2)$.
Scaling back proves the assertion for every $\alpha>0$.
\end{proof}

\begin{remark}[explicit test functions]
The proof is constructive.  For fixed $R$, let
\[
 f_{R,\varepsilon}
 =\frac{\mathbf1_{B(y_R,\varepsilon)}}
        {\mu_2(B(y_R,\varepsilon))}.
\]
Then $\|f_{R,\varepsilon}\|_{L^1(d\mu_2)}=1$ and Fatou gives
\[
 \liminf_{\varepsilon\downarrow0}
 \|P_1f_{R,\varepsilon}\|_{L^1(d\mu_2)}\ge J(R).
\]
Choosing first $R\to\infty$ and then $\varepsilon\downarrow0$ produces an
explicit norm-divergent family.
\end{remark}

\section{What fails in the source proof}

The source's Theorem 2.7 claims boundedness at $p=1$, $\beta=2\alpha$ by
estimating the absolute kernel integral
\[
 I_\alpha^\beta(y)=\int |H_\alpha(y,x)|\dd\mu_\beta(x).
\]
In the cone calculation leading to its equations (2.24)--(2.25), a
difference of exponentials is replaced by a mean-value expression involving
a parameter
\[
 \theta\in(\sqrt{1-\varepsilon^2},1).
\]
The subsequent Kummer estimate treats this as uniform exponential slack
strictly below $1$.  But the mean-value parameter depends on the radial
variables and on $|y|$ and can approach $1$.  Formula
\eqref{exact-column} shows concretely that no uniform slack exists: on a
fixed Gaussian tube around $y_R$, the absolute mass is of order $R$.

This counterexample directly contradicts Theorem 1.4(b), Corollary 1.5,
and Theorem 2.7 of the source at $n=4$, $p=1$.  It disproves Conjecture
1.6(b) as written.  It does \emph{not} decide whether the critical projection
could still be bounded for $1<p<\infty$ in odd dimensions; cancellation in
the input variable can matter away from the $L^1$ endpoint.

\section{Verification and novelty check}

The accompanying script \texttt{verify\_column\_growth.py} performs four
independent checks:
\begin{enumerate}
\item it compares \eqref{kernel} against high-order partial sums of the
zonal-harmonic series;
\item it checks the exact Gaussian exponent cancellation;
\item it quadratures \eqref{exact-column} for increasing $R$;
\item it compares the numerical columns with the explicit restricted-region
lower bound used above.
\end{enumerate}
