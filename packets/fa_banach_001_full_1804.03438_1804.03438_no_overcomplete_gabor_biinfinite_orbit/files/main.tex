\title{No overcomplete lattice Gabor frame is a bi-infinite operator orbit\\
\large A full answer to the Gabor part of the question in arXiv:1804.03438}
\author{Autonomous solve-first investigation, lane 12}
\date{11 August 2026}

\begin{abstract}
Christensen, Hasannasab, and Philipp ask whether an overcomplete Gabor frame
can, after some ordering by $\Z$, be represented as a full orbit
$(T^nf_0)_{n\in\Z}$ of one bounded operator.  Corso later proved
nonexistence for irrational lattice density and for one compact-support
class, leaving general rational density open.  We give a negative answer in
full: no overcomplete one-window lattice Gabor frame has such an ordering.
After canonical tightening, a hypothetical orbit becomes unitary.  Its
threshold-correlation graphs are Cayley graphs of $\Z$ and have linear
growth, whereas a nondegenerate Gabor correlation graph contains a
quadratically growing $\Z^2$ subgraph.  In the remaining rank-one case, the
two graphs have different numbers of connected components.
\end{abstract}

\paragraph{The question and its later refinement.}
For $a,b>0$ and $g\in L^2(\mathbb R)\setminus\{0\}$, write
\[
 \pi(m,n)=M_{mb}T_{na},\qquad
 \G(g,a,b)=\{\pi(m,n)g:(m,n)\in\Z^2\}.
\]
The source paper asks whether there exist overcomplete Gabor frames that,
under an ordering by $\Z$, are $(T^nf_0)_{n\in\Z}$ for a bounded operator
$T$ \cite[p.~4]{CHP}.  Negative powers and the frame property force $T$ to
be boundedly invertible.  Corso subsequently ruled this out when $ab$ is
irrational and when $g$ has support in an interval of length $a$, but
explicitly left the remaining rational-density case open
\cite[pp.~10--11]{C}.

\begin{theorem}[full Gabor answer]\label{thm:main}
Let $\G(g,a,b)$ be an overcomplete frame for $L^2(\mathbb R)$.  There are no
bijection $\varphi:\Z\to\Z^2$, vector $f_0$, and
$T\in GL(L^2(\mathbb R))$ such that
\[
                   \pi(\varphi(n))g=T^nf_0
                   \qquad(n\in\Z).
\]
Thus the answer to the Gabor part of the source question is negative for all
lattice densities and all nonzero $L^2$ windows.
\end{theorem}

\section{Two reductions}

\begin{lemma}[canonical tight orbit]\label{lem:tight}
If the ordering in Theorem~\ref{thm:main} existed and $S$ were the Gabor frame
operator, then, with $\gamma=S^{-1/2}g$,
\[
 t_n:=\pi(\varphi(n))\gamma=U^nt_0,qquad
 U=S^{-1/2}TS^{1/2},                                   \tag{1}
\]
would be an overcomplete Parseval frame generated by a unitary $U$.
\end{lemma}

\begin{proof}
The Gabor frame operator commutes with every lattice shift $\pi(m,n)$: conjugating
the defining frame-operator sum by one shift merely reindexes its rank-one
terms (the projective phases cancel).  Functional calculus therefore gives
$S^{-1/2}\pi(m,n)=\pi(m,n)S^{-1/2}$, proving (1).  The canonical tight frame
is Parseval and remains overcomplete because $S^{-1/2}$ is invertible.

For rank-one operators written $x\otimes x$, strong convergence of the
Parseval frame-operator sum gives
\[
 UU^*=U\Big(\sum_{n\in\Z}t_n\otimes t_n\Big)U^*
      =\sum_{n\in\Z}t_{n+1}\otimes t_{n+1}=I.
\]
Since $U$ is invertible, it is unitary.
\end{proof}

Put
\[
 r_k=\langle t_0,U^kt_0\rangle\qquad(k\in\Z).
\]

\begin{lemma}[decay of unitary-orbit correlations]\label{lem:decay}
For the Bessel unitary orbit in Lemma~\ref{lem:tight}, $r_k\to0$ as
$|k|\to\infty$.  Hence, for every $\varepsilon>0$,
\[
 Q_\varepsilon=\{k\in\Z\setminus\{0\}:|r_k|\ge\varepsilon\}
                                                               \tag{2}
\]
is finite.
\end{lemma}

\begin{proof}
Let $\mu$ be the scalar spectral measure of $U$ at $t_0$.  If
$p(z)=\sum c_kz^k$ is a trigonometric polynomial, the Bessel synthesis bound
gives
\[
 \int_\T|p|^2\,d\mu=\|p(U)t_0\|^2
 \le B\sum_k|c_k|^2=B\int_\T|p|^2\,dm.
\]
Thus $d\mu=w\,dm$ for some $w\in L^\infty(\T)$.  Since
$r_k=\int_\T z^k w(z)\,dm(z)$, the Riemann--Lebesgue lemma proves the claim.
\end{proof}

Time-frequency covariance gives, for $v,\lambda\in\Z^2$,
\[
 |\langle\pi(\lambda)\gamma,\pi(\lambda+v)\gamma\rangle|
 =|a(v)|,
 \qquad a(v):=\langle\gamma,\pi(v)\gamma\rangle.       \tag{3}
\]
Only the magnitude is used, so the Weyl cocycle causes no issue.  Define
\[
 D=\{v\in\Z^2\setminus\{0\}:a(v)\ne0\},\qquad H=\langle D\rangle\le\Z^2.
                                                               \tag{4}
\]
The bijection $\varphi$ in (1) preserves all inner-product magnitudes:
\[
 |a(\varphi(j)-\varphi(i))|=|r_{j-i}|.                \tag{5}
\]

\section{Correlation-graph obstruction}

Regard a family of vectors as a graph by joining two distinct indices exactly
when their inner product is nonzero.  By (3), the Gabor graph has components
equal to the cosets of $H$.  By (5), the unitary-orbit graph has components
equal to the cosets in $\Z$ of
\[
 K=\langle Q\rangle,\qquad
 Q=\{k\in\Z\setminus\{0\}:r_k\ne0\}.                 \tag{6}
\]
Moreover, $\varphi$ is an isomorphism between these graphs.

Suppose first that $\rank H\le1$.  Then $H$ has infinite index in $\Z^2$, so
the Gabor graph has infinitely many connected components.  On the other hand,
$Q\ne\varnothing$: if all off-diagonal $r_k$ vanished, the Parseval orbit
would be an orthonormal basis, contrary to overcompleteness.  Hence
$K=d\Z$ for some $d\ge1$, and the orbit graph has exactly $d$ components.
This contradicts graph isomorphism.

It remains to treat $\rank H=2$.  Choose linearly independent $v,w\in D$ and
fix
\[
 0<\varepsilon<\min\{|a(v)|,|a(w)|\}.
\]
Join indices when their inner-product magnitude is at least $\varepsilon$.
On $\Z^2$ this threshold graph contains the edges with steps $\pm v,\pm w$.
Thus its radius-$R$ ball about any vertex contains
\[
 \{mv+nw:|m|+|n|\le R\},
\]
which has $1+2R(R+1)$ elements because $v,w$ are independent.

Under $\varphi$, this is isomorphic to the threshold graph on $\Z$ with step
set $Q_\varepsilon$ from (2).  Let
$M=\max\{|q|:q\in Q_\varepsilon\}$; the set is nonempty because the
$v$-edge has an image.  Every radius-$R$ ball in this graph lies in an integer
interval of length $2MR$, so it has at most $2MR+1$ vertices.  For large $R$
this contradicts the quadratic lower bound.  Both possible ranks of $H$ are
impossible, completing the proof of Theorem~\ref{thm:main}. \qed

\paragraph{Scope.}
The proof closes the Gabor existence question, including the rational-density
case left in \cite{C}.  The same argument applies to lattice Gabor frames on
$L^2(\mathbb R^d)$, since their index group is $\Z^{2d}$: rank at least two
gives the same growth contradiction, while rank at most one gives the
component contradiction.  The wavelet half of the source question is not
claimed here; the usual $(j,k)$ wavelet family lacks the stationary abelian
$\Z^2$ correlation law used in (3).
