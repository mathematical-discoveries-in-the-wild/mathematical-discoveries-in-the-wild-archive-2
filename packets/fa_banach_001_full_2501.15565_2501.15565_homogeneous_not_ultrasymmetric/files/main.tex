\noindent\textbf{Status: candidate full negative resolution of Remark 2.14.}

\section{The question and the answer}

Boza, K\v{r}epela, and Soria ask whether every $p$-homogeneous
rearrangement-invariant (r.i.) space on $(0,\infty)$ must be ultrasymmetric
for the fundamental function $\varphi(t)=t^{1/p}$
\cite[Remark~2.14]{BKS}.

\begin{center}
\includegraphics[width=0.96\textwidth]{figures/open_problem_crop.png}
\end{center}

The answer is no, already for $p=2$.  For a measurable function $f$, let
$f^*$ be the nonincreasing rearrangement of $|f|$ and put
\[
 f^{**}(t)=\frac1t\int_0^t f^*(u)\,du,
 \qquad
 G_f(s)=e^{s/2}f^{**}(e^s) \quad(s\in\R).
\]
For a nonnegative measurable $g$ on $\R$, define the translation-invariant
Morrey functional
\begin{equation}\label{eq:morrey}
 \Mcal(g)=\sup_I |I|^{-1/2}\int_I g(s)\,ds,
\end{equation}
where the supremum is over all bounded nondegenerate intervals $I\subset\R$.

\begin{theorem}\label{thm:main}
The functional
\begin{equation}\label{eq:Xnorm}
 \|f\|_X:=\Mcal(G_f)
\end{equation}
is a rearrangement-invariant Banach function norm on $(0,\infty)$ and is
exactly $2$-homogeneous:
\[
 \|D_r f\|_X=r^{-1/2}\|f\|_X \qquad(r>0).
\]
Its fundamental function is a constant multiple of $t^{1/2}$, but $X$ is
not ultrasymmetric.  Consequently the question in Remark~2.14 of
\cite{BKS} has a negative answer.
\end{theorem}

\section{Proof intuition}

The change of variables $t=e^s$ turns dilation into translation.  Exact
$2$-homogeneity therefore requires only a translation-invariant norm of the
logarithmic profile $G_f$.  Ultrasymmetry is stronger: Pustylnik's
characterization requires a fully rearrangement-invariant parameter norm on
the logarithmic line \cite{Pus}.

The Morrey functional in \eqref{eq:morrey} remembers where mass sits.  We
construct two profiles from the same multiset of gaps.  Their decreasing
rearrangements are uniformly comparable, but one ordering groups many kernels
in a cluster while the other disperses them into isolated pairs.  Every r.i.
parameter norm must treat the profiles comparably, whereas $\Mcal$ makes
their norms differ by order $\sqrt m$.

\section{The norm and its homogeneity}

\begin{lemma}\label{lem:norm}
The functional in \eqref{eq:Xnorm} satisfies the Banach-function-norm axioms,
is r.i., and obeys
\begin{equation}\label{eq:dilation}
 G_{D_r f}(s)=r^{-1/2}G_f(s+\log r).
\end{equation}
Moreover, for a measurable set $E$ of measure $a$,
\begin{equation}\label{eq:fundamental}
 \|\chi_E\|_X=a^{1/2}\Mcal(k),
 \qquad k(s):=e^{-|s|/2}.
\end{equation}
\end{lemma}

\begin{proof}
The functional depends only on $f^*$, so it is rearrangement invariant.
The standard subadditivity
\[
 (f+g)^{**}(t)\leq f^{**}(t)+g^{**}(t)
\]
gives the triangle inequality, and lattice monotonicity is immediate.
Definiteness and scalar homogeneity are clear.

For the Fatou axiom, suppose $0\leq f_n\uparrow f$.  The identity
\[
 t f_n^{**}(t)
 =\sup_{|A|\leq t}\int_A f_n
\]
and monotone convergence show that $f_n^{**}(t)\uparrow f^{**}(t)$ for every
$t>0$.  Monotone convergence on each interval, followed by the interval
supremum, gives $\|f_n\|_X\uparrow\|f\|_X$.

If $f=\chi_E$ and $|E|=a$, direct calculation gives
\[
 G_f(s)=a^{1/2}k(s-\log a).
\]
Since $k\in L^1(\R)\cap L^\infty(\R)$, $0<\Mcal(k)<\infty$, proving the
finite-characteristic-function axiom and \eqref{eq:fundamental}.

It remains to verify local integral control.  Monotonicity of $f^{**}(t)$ and
of $t f^{**}(t)$ implies, for $|u|\leq1$,
\begin{equation}\label{eq:persistence}
 G_f(s+u)\geq e^{-|u|/2}G_f(s).
\end{equation}
Consequently
\[
 \Mcal(G_f)
 \geq 2^{-1/2}\int_{s-1}^{s+1}G_f(v)\,dv
 \geq \sqrt2e^{-1/2}G_f(s).
\]
For any measurable $E$ with $|E|=a<\infty$, the Hardy--Littlewood
inequality now yields
\[
 \int_E|f|
 \leq\int_0^a f^*(t)\,dt
 =a f^{**}(a)
 =a^{1/2}G_f(\log a)
 \leq \frac{e^{1/2}}{\sqrt2}a^{1/2}\|f\|_X.
\]
This proves the remaining Banach-function-norm axiom.  The usual theorem for
Banach function norms then supplies completeness.

Finally, $(D_r f)^{**}(t)=f^{**}(rt)$, which proves
\eqref{eq:dilation}.  Translation invariance of \eqref{eq:morrey} gives
$\|D_r f\|_X=r^{-1/2}\|f\|_X$.
\end{proof}

\section{Profiles with identical gap data}

Let $S=\{s_1<\cdots<s_N\}$ be a finite set and define the decreasing step
function
\begin{equation}\label{eq:fS}
 f_S(t)=\sum_{i=1}^N e^{-s_i/2}\chi_{(0,e^{s_i})}(t).
\end{equation}
Thus $f_S^*=f_S$.  Introduce the two logarithmic profiles
\[
 H_S(s):=e^{s/2}f_S^*(e^s),
 \qquad
 G_S(s):=e^{s/2}f_S^{**}(e^s).
\]
A direct calculation gives
\begin{equation}\label{eq:profiles}
 H_S(s)=\sum_{s_i\geq s}e^{(s-s_i)/2},
 \qquad
 G_S(s)=\sum_{i=1}^N k(s-s_i).
\end{equation}

Assume all consecutive gaps are at least a fixed $L>0$.  For $s\leq s_N$,
let $s_j$ be the first center at or to the right of $s$ and set
\[
 Q_S(s)=e^{(s-s_j)/2};
\]
put $Q_S(s)=0$ for $s>s_N$.  The minimum-gap assumption and a geometric
series give
\begin{equation}\label{eq:HQ}
 Q_S\leq H_S\leq C_LQ_S,
 \qquad C_L=(1-e^{-L/2})^{-1}.
\end{equation}
If the gaps are $d_i=s_{i+1}-s_i$, then for $0<\lambda<1$,
\begin{equation}\label{eq:distribution}
 |\{Q_S>\lambda\}|
 =2\log(1/\lambda)
  +\sum_{i=1}^{N-1}\min\{d_i,2\log(1/\lambda)\}.
\end{equation}
Indeed, the leftmost half-line contributes $2\log(1/\lambda)$, while the
interval of length $d_i$ immediately preceding $s_{i+1}$ contributes the
minimum appearing in the sum.

It follows that the distribution of $Q_S$ depends only on the \emph{multiset}
of gaps, not their order.  Therefore, whenever $S$ and $T$ have the same gap
multiset and minimum gap at least $L$, \eqref{eq:HQ} implies
\begin{equation}\label{eq:Hstar}
 C_L^{-1}H_T^*\leq H_S^*\leq C_LH_T^*.
\end{equation}

\section{Clustering versus dispersion}

Fix $L\geq4$.  For each integer $m\geq2$, put $R_m=Lm^2$.  Let $S_m$ have
the ordered gap list
\begin{equation}\label{eq:cluster-gaps}
 \underbrace{L,\ldots,L}_{m\ \mathrm{times}},
 \underbrace{R_m,\ldots,R_m}_{m\ \mathrm{times}},
\end{equation}
and let $T_m$ have the alternating gap list
\begin{equation}\label{eq:spread-gaps}
 L,R_m,L,R_m,\ldots,L,R_m.
\end{equation}
Both sets contain $2m+1$ centers and have the same gap multiset.

\begin{lemma}\label{lem:morrey-separation}
There are constants $c_L,C_L'>0$, independent of $m$, such that
\[
 \Mcal(G_{S_m})\geq c_L\sqrt m,
 \qquad
 \Mcal(G_{T_m})\leq C_L'.
\]
\end{lemma}

\begin{proof}
In $S_m$, the first $m+1$ centers have spacing $L$.  The interval extending
one unit beyond the two endpoints of this cluster has length $mL+2$ and
captures at least
$c_0:=\int_{-1}^1k(u)\,du>0$ from each of those kernels.  Hence
\[
 \Mcal(G_{S_m})
 \geq\frac{c_0(m+1)}{\sqrt{mL+2}}
 \geq c_L\sqrt m.
\]

For the dispersed set, first note two elementary estimates valid for any
finite set of centers with minimum gap $L$.  The geometric decay of $k$
gives
\begin{equation}\label{eq:sup-bound}
 \sup_s\sum_i k(s-s_i)\leq A_L,
\end{equation}
and, for every interval $I$,
\begin{equation}\label{eq:int-count}
 \int_I\sum_i k(s-s_i)\,ds
 \leq4\#(S\cap I)+B_L.
\end{equation}
For \eqref{eq:int-count}, the centers inside $I$ contribute at most
$\int_\R k=4$ each, while the centers outside either endpoint form two
geometrically summable tails.

The alternating construction gives, for $\ell=|I|$,
\[
 \#(T_m\cap I)
 \leq\min\{2m+1,\,2\ell/R_m+5\}.
\]
If $\ell\leq1$, \eqref{eq:sup-bound} gives
$\ell^{-1/2}\int_I G_{T_m}\leq A_L$.  If $1<\ell\leq mR_m$,
\eqref{eq:int-count} and the local count give
\[
 \ell^{-1/2}\int_I G_{T_m}
 \leq C_L\left(\frac{\sqrt\ell}{R_m}+\frac1{\sqrt\ell}\right)
 \leq C_L\left(\sqrt{\frac m{R_m}}+1\right).
\]
For $\ell>mR_m$, use the total count instead to obtain
\[
 \ell^{-1/2}\int_I G_{T_m}
 \leq C_L\frac m{\sqrt{mR_m}}+C_L
 \leq C_L'.
\]
Since $R_m=Lm^2$, all bounds are uniform in $m$.
\end{proof}

\section{Failure of ultrasymmetry}

\begin{proof}[Completion of the proof of Theorem~\ref{thm:main}]
Suppose, toward a contradiction, that $X$ is ultrasymmetric.  By
\eqref{eq:fundamental}, its fundamental function is equivalent (indeed equal
up to a constant) to $t^{1/2}$.  Pustylnik's characterization of
ultrasymmetric spaces \cite[Theorems~2.1--2.2]{Pus} then supplies an r.i.
space $E$ on $((0,\infty),dt/t)$ such that
\begin{equation}\label{eq:pustylnik}
 \|f\|_X\asymp\|t^{1/2}f^*(t)\|_E.
\end{equation}
After $t=e^s$, the parameter space is rearrangement invariant on
$(\R,ds)$ and the function on the right of \eqref{eq:pustylnik} is $H_S$ for
$f=f_S$.

The gap multisets of $S_m$ and $T_m$ agree.  Hence \eqref{eq:Hstar} and the
lattice and rearrangement-invariance properties of $E$ imply
\[
 \|H_{S_m}\|_E\asymp_L\|H_{T_m}\|_E
\]
with constants independent of $m$.  Equation \eqref{eq:pustylnik} would
therefore make $\|f_{S_m}\|_X/\|f_{T_m}\|_X$ uniformly bounded.  But
\eqref{eq:Xnorm}, \eqref{eq:profiles}, and
Lemma~\ref{lem:morrey-separation} give
\[
 \frac{\|f_{S_m}\|_X}{\|f_{T_m}\|_X}
 =\frac{\Mcal(G_{S_m})}{\Mcal(G_{T_m})}
 \geq c_L'\sqrt m\longrightarrow\infty.
\]
This contradiction proves that $X$ is not ultrasymmetric.
\end{proof}

\section*{Verification, novelty, and scope}

The counterexample is an ordinary Banach r.i. space, not merely a quasi-Banach
space.  It has exact homogeneity for every real dilation, not only dyadic
dilations.  The obstruction is norm-equivalence invariant: the contradiction
uses the representation of every ultrasymmetric space up to equivalent norm.
Only $p=2$ is needed to refute the universal question; analogous asymmetric
exponential kernels should treat other $1<p<\infty$, but that extension is
not claimed here.

The bounded novelty search covered the exact source question, the article
title and identifier, homogeneous r.i. spaces, ultrasymmetric spaces, and
Pustylnik's representation.  The source was updated in January 2026 and
published in July 2026 with the question still present.  No later answer was
located.  Novelty is therefore provisional pending expert review.
