\title{The missing factor is logarithmic\
\large A sharp solution of the strict small-smoothness Morrey entropy gap}
\author{Autonomous solve-first investigation, lane 12}
\date{13 August 2026}

\begin{document}

\begin{abstract}
Haroske and Skrzypczak left an epsilon gap for entropy numbers of compact
embeddings between proper smoothness Morrey spaces.  In the strict
small-smoothness range we close it: the exact order is
\[
 k^{-\alpha}(\log(e+k))^\alpha,
\]
not the previously suggested pure power.  The lower bound comes from a
large dyadic code with one spike in each mesoscopic block.  The matching
upper bound follows by retaining the finite-level Schuett estimate already
present in the source proof and distributing the entropy budget directly
across wavelet levels.  At the limiting face, the same packing gives the new
lower bound $k^{-D}(\log(e+k))^D$, but the matching endpoint upper bound,
which may depend on the fine indices, is not claimed here.
\end{abstract}

\section{Source question and status}

The source is D.~D.~Haroske and L.~Skrzypczak,
\emph{Entropy numbers of compact embeddings of smoothness Morrey spaces on
bounded domains}, arXiv:1902.04945, later J. Approx. Theory 256 (2020),
105424 \cite{HS}.  Its Theorems 3.13(ii) and 4.4(ii) prove
\begin{equation}\label{eq:old}
 c k^{-\alpha}\le e_k(T)\le c_\varepsilon k^{-\alpha+\varepsilon}
 \qquad(\varepsilon>0),
\end{equation}
and Remark 4.12 asks whether a direct covering argument can close the gap.
The exact printed passage is shown in Figure~\ref{fig:question}.

The result below fully determines the entropy order in the open strict
range.  Because the limiting equality case remains only partially resolved,
this packet is conservatively classified as a substantial partial solution
to the whole of Remark 4.12.

\section{Notation}

Fix $0<p_i\le u_i<\infty$ and $0<q_i\le\infty$, $i=1,2$, and put
\begin{equation}\label{eq:parameters}
 D=\frac1{p_1}-\frac1{p_2},\qquad
 a=1-\frac{p_1}{u_1},\qquad
 \Delta=\frac{s_1-s_2}{d},\qquad
 \beta=\Delta-\frac{p_1}{u_1}D,qquad
 \alpha=\frac{\beta}{a}.
\end{equation}
In the nonclassical situation of the source,
\begin{equation}\label{eq:nonclassical}
 \frac{p_1}{u_1}D>
 \max\left\{0,\frac1{u_1}-\frac1{u_2}\right\},
\end{equation}
so $p_1<p_2$, $p_1<u_1$, and $p_2/u_2>p_1/u_1$.

For completeness, at a fixed dyadic level $j$ let $n=2^{jd}$ and let
$m^{(j)}_{u,p}$ be $\R^n$ with norm
\begin{equation}\label{eq:blocknorm}
 \|x\|_{m^{(j)}_{u,p}}
 =\max_{0\le\nu\le j}\ \max_{Q_{\nu,r}\subset Q}
 N_{\nu}^{\,1/u-1/p}
 \left(\sum_{Q_{j,m}\subset Q_{\nu,r}}|x_m|^p\right)^{1/p},
 \qquad N_\nu=2^{d(j-\nu)}.
\end{equation}
The associated bounded-cube Besov--Morrey sequence norm is
\begin{equation}\label{eq:fullnorm}
 \|x\|_{\widetilde n^\sigma_{u,p,q}}
 =\left\|\left(
 2^{j(\sigma-d/u)}\|x^{(j)}\|_{m^{(j)}_{u,p}}
 \right)_{j\ge0}\right\|_{\ell_q}.
\end{equation}
As in the source, $\sigma_i=s_i+d/2$, so only their difference matters.

\section{Proof intuition}

The published lower bound treats level $j$ as if it had about $n^a$
effective coordinates.  That count is right, but each effective coordinate
can be placed in about $n^{1-a}$ different fine cubes.  Coding these choices
produces $\exp(cn^a\log n)$, rather than merely $\exp(cn^a)$, separated
vectors.  This forces the logarithm in the answer.

For the upper bound one should not replace the finite-level entropy estimate
by an $L_{r,\infty}^{(e)}$ ideal norm, because taking $1/r>D$ is exactly where
the epsilon is introduced.  The source proves a sharper Schuett estimate
immediately before that step.  Its transition also occurs at
$n^a\log n$.  In the strict range $\beta<aD$, entropy budgets assigned to
successive lower levels form a geometric series.  This gives a cover of
radius $N^{-\beta}$ using $O(N^a\log N)$ bits, precisely matching the code.

\section{The strict-range theorem}

\begin{theorem}[sharp logarithmic order]\label{thm:main}
Let $\Omega\subset\R^d$ be any bounded domain, let
$\mathcal A\in\{\mathcal N,\mathcal E\}$, and assume
\eqref{eq:nonclassical} and
\begin{equation}\label{eq:strict}
 \frac{p_1}{u_1}D<\Delta<D.
\end{equation}
For the compact embedding
\[
 T:\mathcal A^{s_1}_{u_1,p_1,q_1}(\Omega)
 \longrightarrow
 \mathcal A^{s_2}_{u_2,p_2,q_2}(\Omega)
\]
one has, for every $k\in\N$,
\begin{equation}\label{eq:answer}
 e_k(T)\asymp
 k^{-\alpha}(\log(e+k))^\alpha,
 \qquad
 \alpha=\frac{u_1}{u_1-p_1}
 \left(\Delta-\frac{p_1}{u_1}D\right).
\end{equation}
The constants may depend on the fixed parameters and on $\Omega$, but not on
$k$.
\end{theorem}

We prove the sequence-space statement first.

\begin{lemma}[finite-level upper estimate]\label{lem:blockupper}
Set
\[
 \gamma=\frac1{u_2}-\frac{p_1}{u_1p_2}>0,
 \qquad
 S_j:m^{(j)}_{u_1,p_1}\longrightarrow m^{(j)}_{u_2,p_2}.
\]
Uniformly in $n=2^{jd}$ and $m\ge1$,
\begin{align}
 e_m(S_j)&\lesssim
 n^\gamma\min\left\{1,
 \left(\frac{n^a\log(e+n/m)}m\right)^D\right\},
 &&1\le m\lesssim n,\label{eq:blockmiddle}\\
 e_m(S_j)&\lesssim
 n^{1/u_2-1/u_1}2^{-c m/n},
 &&m\gtrsim n.\label{eq:blocklarge}
\end{align}
Consequently the level operator induced by
$\widetilde n^{\sigma_1}_{u_1,p_1,q_1}\to
\widetilde n^{\sigma_2}_{u_2,p_2,q_2}$ satisfies
\begin{align}
 e_m(T_j)&\lesssim
 n^{-\beta}\min\left\{1,
 \left(\frac{n^a\log(e+n/m)}m\right)^D\right\},
 &&m\lesssim n,\label{eq:weightedmiddle}\\
 e_m(T_j)&\lesssim n^{-\Delta}2^{-c m/n},
 &&m\gtrsim n.\label{eq:weightedlarge}
\end{align}
\end{lemma}

\begin{proof}
Equation (3.35) in Step 1 of Proposition 3.12 of the source gives, for the
renormalized target block $m^*_{u_2,p_2}$,
\[
 e_{2m-1}\big(m^{(j)}_{u_1,p_1}\to m^*_{u_2,p_2}\big)
 \lesssim n^{aD}
 \left(\frac{\log(1+n/m)}m\right)^D
\]
in the Schuett range.  Equation (3.32) says that returning to the original
target multiplies entropy numbers by $n^\gamma$.  Taking the minimum with
$\|S_j\|\lesssim n^\gamma$ yields \eqref{eq:blockmiddle}; the last branch of
(3.35), equivalently the standard finite-dimensional volumetric estimate,
gives \eqref{eq:blocklarge}.

The ratio of the level weights in \eqref{eq:fullnorm} is
$n^{-(\Delta-1/u_1+1/u_2)}$.  Since
\[
 -\Delta+\frac1{u_1}-\frac1{u_2}+\gamma=-\beta,
 \qquad
 -\Delta+\frac1{u_1}-\frac1{u_2}
 +\frac1{u_2}-\frac1{u_1}=-\Delta,
\]
the weighted estimates follow.
\end{proof}

\begin{lemma}[multilevel upper bound]\label{lem:globalupper}
Under \eqref{eq:strict}, for $N=2^{Jd}$,
\begin{equation}\label{eq:dyadicupper}
 e_{C N^a(1+J)}
 \big(\widetilde n^{\sigma_1}_{u_1,p_1,q_1}	o
      \widetilde n^{\sigma_2}_{u_2,p_2,q_2}\big)
 \lesssim N^{-\beta}.
\end{equation}
\end{lemma}

\begin{proof}
Choose
\begin{equation}\label{eq:eta}
 0<\eta<D-\Delta=aD-\beta
\end{equation}
and put $R=N^{-\beta}$.  The tail $j>J$ is covered by the zero center:
from \eqref{eq:weightedmiddle} with $m=1$ and the fact that every normalized
source level is at most one,
\[
 \left\|(T_jx^{(j)})_{j>J}\right\|_{\ell_{q_2}}
 \lesssim
 \left\|(2^{-jd\beta})_{j>J}\right\|_{\ell_{q_2}}
 \lesssim R,
\]
with the evident supremum interpretation when $q_2=\infty$.

For $j\le J$, write $r=J-j$, $n_j=N2^{-dr}$, and prescribe the level error
\begin{equation}\label{eq:levelerror}
 R_j=R2^{-\eta dr}.
\end{equation}
Let
\[
 c_0=a-\frac{\beta+\eta}{D}>0.
\]
Whenever the following number is at most a fixed multiple of $n_j$, choose
\begin{equation}\label{eq:budgetmiddle}
 m_j=\left\lceil
 C N^a(1+j)2^{-c_0dr}
 \right\rceil.
\end{equation}
Since $\log(e+n_j/m_j)\lesssim1+j$, the bracket in
\eqref{eq:weightedmiddle} is at most a constant times
$2^{-(\beta+\eta)dr/D}$.  Hence $e_{m_j}(T_j)\lesssim R_j$.
Moreover,
\begin{equation}\label{eq:middlebudget}
 \sum m_j\lesssim
 N^a\sum_{r\ge0}(1+J-r)2^{-c_0dr}
 \lesssim N^a(1+J).
\end{equation}

It remains to audit the lower levels where \eqref{eq:budgetmiddle} exceeds
$n_j$.  There use \eqref{eq:weightedlarge} and take
\begin{equation}\label{eq:budgetlarge}
 m_j=\left\lceil Cn_j
 \left(1+\log_+\frac{n_j^{-\Delta}}{R_j}\right)\right\rceil.
\end{equation}
This again makes $e_{m_j}(T_j)\lesssim R_j$.  The condition that
\eqref{eq:budgetmiddle} exceeds $n_j$ implies
\[
 n_j\lesssim
 N^\kappa(1+J)^{1/(1-c_0)},
 \qquad
 \kappa=\frac{a-c_0}{1-c_0}<a.
\]
The $n_j$ are geometric, while the logarithm in
\eqref{eq:budgetlarge} is $O(1+J)$.  Thus the sum of all budgets in this
lower-level part is
\[
 O\big(N^\kappa(1+J)^{2+1/(1-c_0)}\big)
 =o\big(N^a(1+J)\big).
\]

Take the Cartesian product of the chosen level covers.  It uses at most
$2^{C N^a(1+J)}$ centers.  By \eqref{eq:levelerror}, its target radius is
\[
 \left\|(R_j)_{j\le J}\right\|_{\ell_{q_2}}
 \lesssim R;
\]
adding the tail changes only the constant.  This proves
\eqref{eq:dyadicupper}.
\end{proof}

\begin{lemma}[dyadic code lower bound]\label{lem:lower}
Under the nonclassical hypotheses and merely $\beta>0$,
\begin{equation}\label{eq:dyadiclower}
 e_{c n^a\log n}
 \big(\widetilde n^{\sigma_1}_{u_1,p_1,q_1}	o
      \widetilde n^{\sigma_2}_{u_2,p_2,q_2}\big)
 \gtrsim n^{-\beta},
 \qquad n=2^{jd},
\end{equation}
for all sufficiently large $j$.
\end{lemma}

\begin{proof}
Put $\nu=\lfloor aj\rfloor$, $s=2^{d\nu}$, and
$L=2^{d(j-\nu)}$, so $s\asymp n^a$, $L\asymp n^{1-a}$, and $n=sL$.
Partition the fine cubes at level $j$ into the $s$ dyadic cubes at level
$\nu$.  For each word $\omega\in\{1,\ldots,L\}^s$, let $x^\omega$ be the
indicator that chooses the $\omega_t$-th fine cube in parent $t$.

Every $x^\omega$ lies in the unit ball of $m^{(j)}_{u_1,p_1}$.  Indeed, a
dyadic block containing $M$ fine cubes contains at most $M^a$ chosen cubes.
For a block below the parent scale this count is at most one.  For a block
above it the assertion is equivalent to
$2^{d(\nu-\mu)}\le2^{ad(j-\mu)}$, which follows from $\nu\le aj$.
Substitution in \eqref{eq:blocknorm} proves the claim.

A greedy code in the alphabet of size $L$ gives a set
$\mathcal C\subset\{1,\ldots,L\}^s$ with Hamming distance at least $s/2$ and
\begin{equation}\label{eq:codesize}
 |\mathcal C|\ge2^{c s\log_2L}.
\end{equation}
For reference, a Hamming ball of radius $s/2$ has at most
$2^sL^{s/2}$ words, so the elementary greedy deletion argument proves
\eqref{eq:codesize}.

If $\omega,\omega'\in\mathcal C$ are distinct, their indicators differ in
at least $s$ fine coordinates.  Testing the target norm on the whole cube
therefore gives
\begin{equation}\label{eq:separationblock}
 \|x^\omega-x^{\omega'}\|_{m^{(j)}_{u_2,p_2}}
 \gtrsim n^{1/u_2-1/p_2}s^{1/p_2}asymp n^\gamma.
\end{equation}
Multiply every vector by the source normalization
$n^{-(\sigma_1/d-1/u_1)}$.  They now lie in the unit ball of the full source
sequence space and are supported on one level.  After the target weight is
applied, \eqref{eq:separationblock} becomes
\[
 n^{-\Delta+1/u_1-1/u_2}n^\gamma=n^{-\beta}.
\]
The packing cardinality is $2^{c n^a\log n}$, proving
\eqref{eq:dyadiclower} by the definition of entropy numbers.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
The quantities $K_J=2^{Jda}(1+J)$ have bounded successive ratios.  Inverting
\eqref{eq:dyadicupper} and \eqref{eq:dyadiclower}, and using monotonicity of
entropy numbers between adjacent $K_J$, yields
\[
 e_k\big(\widetilde n^{\sigma_1}_{u_1,p_1,q_1}	o
          \widetilde n^{\sigma_2}_{u_2,p_2,q_2}\big)
 \asymp k^{-\beta/a}(\log(e+k))^{\beta/a}.
\]
Indeed, $K_J\asymp n^a\log n$ and
$n^{-\beta}\asymp K_J^{-\beta/a}(\log K_J)^{\beta/a}$.

The two wavelet-factorization diagrams in Section 4 of the source transfer
both entropy inequalities between bounded-cube sequence spaces and
Besov--Morrey spaces on every bounded domain.  Finally, the elementary
embeddings
\[
 \mathcal N^s_{u,p,\min(p,q)}
 \hookrightarrow\mathcal E^s_{u,p,q}
 \hookrightarrow\mathcal N^s_{u,p,\infty}
\]
and multiplicativity of entropy numbers give the same two-sided order for
the Triebel--Lizorkin--Morrey scale.  The formula is independent of the fine
indices, so the two sandwiches agree.
\end{proof}

\section{Endpoint consequence and limitation}

\begin{proposition}\label{prop:endpoint}
Under \eqref{eq:nonclassical}, if $\Delta=D$, then for both function-space
scales and every bounded domain,
\begin{equation}\label{eq:endpoint}
 e_k(T)\gtrsim k^{-D}(\log(e+k))^D.
\end{equation}
In particular, $e_k(T)\asymp k^{-D}$ is impossible.
\end{proposition}

\begin{proof}
Here $\beta=aD$.  Lemma~\ref{lem:lower} and the same adjacent-level
inversion give \eqref{eq:endpoint}; the wavelet transfer is unchanged.
\end{proof}

At this equality face, $\beta=aD$, so the geometric factor in the budget
\eqref{eq:budgetmiddle} vanishes.  Many levels can contribute at the same
order, and a sharp cover may depend on $q_1,q_2$, just as anticipated in
Remark 4.7 of the source.  The packet therefore makes no matching endpoint
upper-bound claim.

\section{Verification and novelty audit}

The following points were checked independently during the solve-first pass.

\begin{itemize}
\item The exponent identities in Lemma~\ref{lem:blockupper} were expanded
symbolically by hand in both directions; the transition at $m=n$ agrees with
the volumetric branch because $\beta+(1-a)D=\Delta$.
\item The source-ball audit in Lemma~\ref{lem:lower} checks every dyadic
scale, not only the whole cube.  The target separation uses a legitimate
whole-cube term in the supremum.
\item The upper budget was split exactly where the Schuett-range allocation
exceeds the actual dimension.  The strict inequality $\Delta<D$ is used
precisely to obtain $c_0>0$ and $\kappa<a$.
\item No numerical experiment is used as proof.
\item On 13 August 2026, searches of the run indexes and bounded web searches
for the exact title, the phrase ``seal the epsilon-gap'', theorem numbers,
Morrey sequence-space entropy, the authors, arXiv:1904.04619, and later
Morrey-embedding work found no prior closure of this question.  The search
did find the source journal version and later nuclearity/survey papers, none
of which supplies this logarithmic theorem.
\end{itemize}
