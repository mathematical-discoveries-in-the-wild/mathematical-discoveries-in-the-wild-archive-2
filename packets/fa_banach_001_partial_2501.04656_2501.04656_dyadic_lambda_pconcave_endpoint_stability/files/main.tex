\author{Autonomous proof-discovery packet for arXiv:2501.04656}

\begin{abstract}
Figalli, van Hintum, and Tiba ask whether the constant in their sharp
Borell--Brascamp--Lieb stability theorem can be made to depend on the mixing
weight as $\lambda^{-1/2}$.  We prove the conjectured estimate for every
dyadic $\lambda=2^{-m}$ when the endpoint carrying weight $1-\lambda$ is
already $p$-concave.  The mechanism is an exact semigroup identity for
$p$-sup convolution.  Homothetic probability densities on cubes show that
the power $\lambda^{-1/2}$ is necessary and, additionally, that the constant
must blow up at least as $(np+1)^{-1/2}$ when $p\downarrow-1/n$.
\end{abstract}

\section{Source question and notation}

For $a,b\geq0$, the source uses the zero convention
\[
 M_{t,p}(a,b)=0\quad\text{if }ab=0,
\]
and, when $ab>0$, the usual weighted power mean
\[
 M_{t,p}(a,b)=
 \begin{cases}
 (ta^p+(1-t)b^p)^{1/p},&p\ne0,\\
 a^t b^{1-t},&p=0.
 \end{cases}
\]
Write
\[
 S_{t,p}(u,v)(z)
 :=\sup_{z=tx+(1-t)y}M_{t,p}(u(x),v(y)).
\]
Theorem 1.6 of Figalli--van Hintum--Tiba~\cite{FVHT2025} proves, for
$p>-1/n$, equal-mass $f,g$, and $h\geq S_{\lambda,p}(f,g)$ with relative
integral deficit $\delta$, that $f,g$ are within
$O_{n,\lambda,p}(\sqrt\delta)$ of a common $p$-concave function, up to a
translation.  Remark 1.8 asks whether this can be sharpened to
\[
 O_{n,p}\!\left(\sqrt{\frac{\delta}{\lambda}}\right),
 \qquad 0<\lambda\leq\frac12.
\]
The exact source passage is reproduced in Figure~\ref{fig:source}.

\section{The dyadic endpoint theorem}

\begin{theorem}[Sharp dyadic dependence]\label{thm:main}
Fix $n\geq1$ and $p>-1/n$.  There is a constant $C_{n,p}$ with the
following property.  Let $\lambda=2^{-m}$, $m\geq1$, and let nonnegative
integrable $f,g,h$ on $\R^n$ satisfy
\[
 \int f=\int g=1,\qquad
 h\geq S_{\lambda,p}(f,g),\qquad
 \int h=1+\delta.
\]
If $g$ is $p$-concave, then there are a $p$-concave function $\ell$ and
$v\in\R^n$ such that
\[
 \norm{f-\ell}+\norm{g(\,\cdot-v)-\ell}
 \leq C_{n,p}\sqrt{\frac{\delta}{\lambda}}.
\]
The constant is independent of $m$.
\end{theorem}

The added hypothesis is one-sided: no concavity is required of $f$.  By
swapping the two inputs, the reflected statement holds at weight
$1-2^{-m}$ when $f$ is $p$-concave.

\section{Exact semigroup identity}

\begin{lemma}\label{lem:semigroup}
If $v$ is $p$-concave and $r,t\in(0,1)$, then
\[
 S_{r,p}\bigl(S_{t,p}(u,v),v\bigr)=S_{rt,p}(u,v).
\]
\end{lemma}

\begin{proof}
Consider a positive term in the nested supremum.  It comes from
\[
 w=tx+(1-t)y_1,qquad z=rw+(1-r)y_2.
\]
Put
\[
 \bar y=\frac{r(1-t)y_1+(1-r)y_2}{1-rt},
 \qquad
 \theta=\frac{r(1-t)}{1-rt}.
\]
Then $z=rtx+(1-rt)\bar y$.  For $p\ne0$, the $p$th-power coefficients in
the nested mean are
\[
 rt\,u(x)^p+r(1-t)v(y_1)^p+(1-r)v(y_2)^p.
\]
The $p$-concavity inequality
$v(\bar y)\geq M_{\theta,p}(v(y_1),v(y_2))$, with the reversal caused by
raising to a negative power when $p<0$, shows that this nested value is at
most $M_{rt,p}(u(x),v(\bar y))$.  The same calculation with logarithms
handles $p=0$.  Terms containing a zero input vanish by convention.
Taking suprema proves ``$\leq$''.

Conversely, in any representation $z=rtx+(1-rt)y$, use $y_1=y_2=y$ and
$w=tx+(1-t)y$.  Monotonicity of the means and the identity
\[
 M_{r,p}\bigl(M_{t,p}(a,b),b\bigr)=M_{rt,p}(a,b)
\]
give the reverse inequality, including the zero cases.
\end{proof}

\section{Proof of the stability theorem}

Set
\[
 u_0=S_{1/2,p}(f,g),\qquad
 u_j=S_{1/2,p}(u_{j-1},g)\quad(1\leq j\leq m-1).
\]
Lemma~\ref{lem:semigroup} gives
\[
 u_j=S_{2^{-j-1},p}(f,g),
 \qquad u_{m-1}=S_{\lambda,p}(f,g)\leq h.
\]
Let $A_j=\int u_j$ and put
\[
 q=\frac{p}{np+1}.
\]
The Borell--Brascamp--Lieb inequality, applied at each midpoint step, yields
in the extended sense
\[
 A_j\geq M_{1/2,q}(A_{j-1},1).
\]
Backward iteration therefore gives
\begin{align*}
 A_0^q-1
 &\leq 2^{m-1}(A_{m-1}^q-1),&&q>0,\\
 \log A_0
 &\leq 2^{m-1}\log A_{m-1},&&q=0,\\
 1-A_0^q
 &\leq 2^{m-1}(1-A_{m-1}^q),&&q<0.
\end{align*}
Since $A_{m-1}\leq1+\delta$ and $2^{m-1}=1/(2\lambda)$, all three cases
imply, whenever $\delta/\lambda$ is bounded by a sufficiently small constant
depending only on $n,p$,
\[
 A_0-1\leq C'_{n,p}\frac{\delta}{\lambda}.
\]
(For $q<0$, the same inequalities first force $A_0<\infty$.)

Now apply Theorem 1.6 of the source only at the fixed weight $1/2$ to the
triple $(f,g,u_0)$.  Its deficit is $A_0-1$, so it supplies a $p$-concave
$\ell$ and a translation $v$ with
\[
 \norm{f-\ell}+\norm{g(\,\cdot-v)-\ell}
 \leq C''_{n,p}\sqrt{A_0-1}
 \leq C_{n,p}\sqrt{\frac{\delta}{\lambda}}.
\]
If $\delta/\lambda$ is not small, take $\ell=0$: the left side is $2$ and
can be absorbed by enlarging $C_{n,p}$.  This proves
Theorem~\ref{thm:main}.

\section{Sharpness and endpoint blow-up}

\begin{proposition}\label{prop:sharp}
Let $Q=[-1/2,1/2]^n$, $a=1+\varepsilon$, and
\[
 f_a=a^{-n}\one_{aQ},\qquad g=\one_Q.
\]
For $0<\lambda\leq1/2$, let $h=S_{\lambda,p}(f_a,g)$ and
$\delta=\int h-1$.  Then, as $\varepsilon\downarrow0$,
\[
 \delta=\frac{n(np+1)}2\lambda(1-\lambda)\varepsilon^2
        +O_{n,p,\lambda}(\varepsilon^3),
\]
whereas
\[
 \inf_{\ell,v}
 \bigl(\norm{f_a-\ell}+\norm{g(\,\cdot-v)-\ell}\bigr)
 =2(1-a^{-n})=2n\varepsilon+O_n(\varepsilon^2).
\]
The infimum is over all $p$-concave $\ell$ and all $v\in\R^n$.
Consequently, the factor $\lambda^{-1/2}$ is optimal, and any constant in
the conjectured estimate satisfies
\[
 C_{n,p}\geq 2\sqrt{\frac{2n}{np+1}}
\]
in the limit $\lambda\downarrow0$.
\end{proposition}

\begin{proof}
The sup convolution is constant on
$(\lambda a+1-\lambda)Q=(1+\lambda\varepsilon)Q$.  Thus, for $p\ne0$,
\[
 1+\delta=(1+\lambda\varepsilon)^n
 \bigl(\lambda(1+\varepsilon)^{-np}+1-\lambda\bigr)^{1/p},
\]
while for $p=0$,
\[
 1+\delta=(1+\lambda\varepsilon)^n(1+\varepsilon)^{-n\lambda}.
\]
Taylor expansion gives the stated quadratic term.

For every $\ell,v$, the triangle inequality gives a lower bound by
$\norm{f_a-g(\,\cdot-v)}$.  The overlap of $aQ$ with any translate of $Q$
has volume at most one, with equality when the smaller cube is contained in
the larger.  Hence this distance is at least $2(1-a^{-n})$.  Equality is
attained with $v=0$ and $\ell=f_a$, which is $p$-concave under the source's
zero convention.  Dividing the two asymptotics yields
\[
 \frac{2(1-a^{-n})}{\sqrt{\delta/\lambda}}
 \longrightarrow
 2\sqrt{\frac{2n}{(np+1)(1-\lambda)}}.
\]
This proves both conclusions.
\end{proof}

\section{Proof intuition, scope, and upgrade audit}

\paragraph{Proof intuition.}
Think of $S_{t,p}(f,g)$ as mixing $f$ with a profile $g$ for time $t$.
Mixing the result with the same $p$-concave profile again creates two points
of $g$.  Concavity lets us collapse those points to their barycenter with no
loss, so the two mixing times multiply exactly.  A tiny deficit after many
half-mixings can then be transported backward to the first midpoint; the
source's already-proved midpoint stability finishes the argument.

The unrestricted conjecture is not claimed.  Six focused upgrade routes
were pursued.  Arbitrary $\lambda$ factors as $2^{-k}t$ with
$t\in(1/4,1/2]$, but this needs a stability constant known to be uniform on
that compact interval, which is not stated in the source.  Without endpoint
concavity, only
$S_{r,p}(S_{t,p}(f,g),g)\geq S_{rt,p}(f,g)$ remains, the wrong direction for
backward deficit control.  An approximate-concavity bootstrap is circular
because sup convolution is not $L^1$-Lipschitz.  Auditing the source's
multistage transport proof did not isolate compact-uniform constants without
rebuilding it.  Direct transport and level-set routes found no replacement
for the missing barycentric collapse.  Finally, the cube computation rules
out improving either the weight exponent or the endpoint blow-up rate.

Bounded primary-source searches through 11 August 2026 found no later proof
of the full conjecture and no prior statement of the dyadic endpoint result.
This is a novelty check, not a guarantee of novelty.
