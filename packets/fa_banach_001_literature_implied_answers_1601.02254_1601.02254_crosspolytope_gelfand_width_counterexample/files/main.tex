\begin{abstract}
Question 1.1 of Giannopoulos--Hioni--Tsolomitis asks whether every
volume-one isotropic convex body $K\subset\R^n$ has
\[
 R(K\cap F)\le C\sqrt{n/k}\,\sqrt n L_K=C\,nL_K/\sqrt k
\]
for a random codimension-$k$ subspace $F$, with one universal constant.
The answer is negative.  For the volume-normalized cross-polytope, the
sharp lower bound for the Gelfand widths of
$\ell_1^n\hookrightarrow\ell_2^n$ forces an additional
$\sqrt{1+\log(n/k)}$ factor for \emph{every} codimension-$k$ subspace.
Taking $k=\lfloor\sqrt n\rfloor$ contradicts the proposed estimate.
\end{abstract}

\section{The question}

The source asks for an absolute $C$ such that, for every isotropic
convex body $K\subset\R^n$ and every $1\le k\le n-1$, a random
$F\in G_{n,n-k}$ satisfies
\begin{equation}\label{eq:question}
 R(K\cap F)\le C\frac{nL_K}{\sqrt k}.
\end{equation}
Here $R(A)=\sup\{\lVert x\rVert_2:x\in A\}$.

\section{The width input}

For $0\le m<n$, define the $m$th Gelfand width by
\[
 d^m(B_1^n,\ell_2^n)
 :=\inf_{A\in\R^{m\times n}}
       \sup\{\lVert x\rVert_2:x\in B_1^n\cap\ker A\}.
\]
The sharp classical estimate of Garnaev--Gluskin, in the convenient
form of Foucart--Pajor--Rauhut--Ullrich
\cite[Theorem 1.1]{FPRU}, says that there is a universal $c>0$ such
that
\begin{equation}\label{eq:gelfand}
 d^m(B_1^n,\ell_2^n)
 \ge c\min\left\{1,
       \sqrt{\frac{1+\log(n/m)}{m}}\right\}
 \qquad(1\le m<n).
\end{equation}
Only this lower bound is needed below.  Notice that it is an infimum
over all kernels.  Consequently, if $F$ has codimension $m$ and
$A$ is any rank-$m$ map with $\ker A=F$, then
\begin{equation}\label{eq:everyF}
 R(B_1^n\cap F)\ge d^m(B_1^n,\ell_2^n).
\end{equation}

\section{Normalization of the cross-polytope}

Let
\[
 a_n:=\left(\frac{n!}{2^n}\right)^{1/n},
 \qquad K_n:=a_n B_1^n.
\]
Since $|B_1^n|=2^n/n!$, the body $K_n$ has volume one.  Sign and
coordinate-permutation symmetries show that its barycenter is zero and
its covariance matrix is a scalar multiple of the identity, so it is
isotropic.  We record the scalar exactly.

\begin{lemma}\label{lem:isotropic}
The isotropic constant of $K_n$ satisfies
\[
 L_{K_n}^2=\frac{2a_n^2}{(n+1)(n+2)}.
\]
\end{lemma}

\begin{proof}
On the positive-orthant simplex
$\Delta_n=\{x_i\ge0:\sum_i x_i\le1\}$, the Dirichlet integral gives
\[
 \int_{\Delta_n}x_1^2\,dx=\frac{2}{(n+2)!}.
\]
After summing over the $2^n$ orthants and dividing by
$|B_1^n|=2^n/n!$, one obtains
\[
 \frac1{|B_1^n|}\int_{B_1^n}x_1^2\,dx
 =\frac{2}{(n+1)(n+2)}.
\]
The mixed second moments vanish by sign symmetry, and scaling by $a_n$
proves the claim.
\end{proof}

In particular,
\begin{equation}\label{eq:rhs}
 \frac{nL_{K_n}}{\sqrt k}
 =\frac{a_n}{\sqrt k}
   \frac{\sqrt2\,n}{\sqrt{(n+1)(n+2)}}
 \le \frac{\sqrt2\,a_n}{\sqrt k}.
\end{equation}

\section{Counterexample}

\begin{theorem}\label{thm:counterexample}
There is no universal constant $C$ for which \eqref{eq:question}
holds.  More precisely, if $k_n=\lfloor\sqrt n\rfloor$, then for all
sufficiently large $n$ and every $F\in G_{n,n-k_n}$,
\[
 \frac{R(K_n\cap F)}{nL_{K_n}/\sqrt{k_n}}
 \ge c'\sqrt{1+\log(n/k_n)},
\]
where $c'>0$ is universal.  Hence the ratio tends to infinity.
\end{theorem}

\begin{proof}
Homogeneity, \eqref{eq:everyF}, and \eqref{eq:gelfand} give, once
$n$ is large enough that the second branch of the minimum applies,
\[
 R(K_n\cap F)
 =a_nR(B_1^n\cap F)
 \ge c a_n\sqrt{\frac{1+\log(n/k_n)}{k_n}}.
\]
Dividing this inequality by \eqref{eq:rhs} yields
\[
 \frac{R(K_n\cap F)}{nL_{K_n}/\sqrt{k_n}}
 \ge \frac{c}{\sqrt2}\sqrt{1+\log(n/k_n)}.
\]
Since $k_n=\lfloor\sqrt n\rfloor$, the last expression diverges.
For any proposed constant $C$, choose $n$ so large that it exceeds
$C$.  Then \eqref{eq:question} fails for every codimension-$k_n$
subspace, and therefore cannot hold for a random one.
\end{proof}

\section{Remarks on scope and novelty}

The argument answers only Question 1.1 of the source; it does not
address the separate question about the existence of a dimension-free
sub-Gaussian direction.  The width estimate predates the source
question, but bounded searches of the run indexes and the arXiv-facing
literature did not find an explicit application identifying the
volume-normalized cross-polytope as a counterexample.  The mathematical
content of this packet is the short normalization-and-width deduction.
