\begin{center}
\fcolorbox{navy}{softblue}{\begin{minipage}{0.91\linewidth}
\textbf{Outcome.} The proposed replacement of strong $L^1$ control by the
weak $c$-norm is false as stated: it fails at $p=q=1$ on every rank-one
noncompact symmetric space.  Nevertheless it is true for every
$1<p<2$ and every strict interior line $p<q<p'$.  For $1<p<2$, only the two
non-unitary edges $q=p,p'$ remain unresolved.
\end{minipage}}
\end{center}

\section*{1. The exact question}

Let $\X=G/K$ be a rank-one Riemannian symmetric space of noncompact type,
$n=\dim\X$, and
\[
 \dd\mu(\lambda)=|c(\lambda)|^{-2}\dd\lambda,
 \qquad \rho_q=(2/q-1)\rho.
\]
For $1\le q<\infty$ set
\[
 \mathcal F_qf(\lambda)=
 \left(\int_K|\widetilde f(\lambda+i\rho_q,k)|^q\dd k\right)^{1/q}
 \frac{|\lambda+i\rho_q|}{|\lambda+i\rho_q+i2\rho|}.
\]
Theorem 1.11 of the source assumes $u\in L^1(\mu)$ and, for
$1\le p\le2$, $p\le q\le p'$, proves
\begin{equation}\label{eq:paley}
 \|\mathcal F_qf\|_{L^p(u^{2-p}\dd\mu)}
 \le C_{p,q}\|u\|_{L^1(\mu)}^{2/p-1}\|f\|_{L^p(\X)}.
\end{equation}
Remark 5.2 asks whether one may instead use
\[
 \|u\|_{1,\infty;\mu}
 :=\sup_{\alpha>0}\alpha\,\mu\{u>\alpha\}<\infty.
\]

\begin{theorem}[Sharp split for the proposed upgrade]\label{thm:split}
On every rank-one $\X$:
\begin{enumerate}
\item The weak-weight version of \eqref{eq:paley} is false for $p=q=1$.
Thus the all-exponent question in Remark 5.2 has a negative answer.
\item If $1<p<2$ and $p<q<p'$, then
\begin{equation}\label{eq:positive}
 \|\mathcal F_qf\|_{L^p(u^{2-p}\dd\mu)}
 \le C_{p,q}\|u\|_{1,\infty;\mu}^{2/p-1}\|f\|_{L^p(\X)}.
\end{equation}
\end{enumerate}
\end{theorem}

\paragraph{Proof intuition.}
The endpoint fails at the critical scale.  Plancherel balls have volume
$\asymp R^n$, so $u(\lambda)=(1+|\lambda|)^{-n}$ is weak $L^1$ but has
logarithmically divergent weighted mass.  Shrinking approximate identities
have Fourier transform close to one on balls of radius $R$ and detect that
divergence.  In the strict interior, the source's pointwise restriction and
Hausdorff--Young estimates real-interpolate to the sharper Lorentz target
$L^{p',p}$; this is exactly the space dual to the power of the weak weight.

\section*{2. Endpoint counterexample}

The source records the rank-one estimate
\begin{equation}\label{eq:c}
 |c(\lambda)|^{-2}\asymp
 \lambda^2(1+|\lambda|)^{n-3}.
\end{equation}
Take
\[
 u(\lambda)=(1+|\lambda|)^{-n}.
\]
For $0<\alpha<1$, the set $\{u>\alpha\}$ is contained in an interval of
radius $\alpha^{-1/n}$.  Equation \eqref{eq:c} gives
\[
 \mu\{u>\alpha\}\le C\alpha^{-1},
 \qquad\text{so}\qquad \|u\|_{1,\infty;\mu}<\infty.
\]
On the other hand, again by \eqref{eq:c},
\begin{equation}\label{eq:log}
 \int_1^R u(\lambda)\dd\mu(\lambda)
 \ge c\int_1^R\lambda^{-n}\lambda^{n-1}\dd\lambda
 =c\log R.
\end{equation}

Choose $f_R\in C_c^\infty(\X)$ nonnegative with
\[
 \|f_R\|_1=1,
 \qquad \supp f_R\subset B(o,R^{-1}).
\]
The Iwasawa projection satisfies
$|H(x^{-1}k)|\le d(o,x)$.  At the upper boundary $q=1$,
\[
 \widetilde f_R(\lambda+i\rho,k)
 =\int_\X f_R(x)e^{(i\lambda-2\rho)H(x^{-1}k)}\dd x.
\]
Using $|e^z-1|\le e^{|z|}|z|$, choose a fixed $b>0$ small enough.  Uniformly
in $k$ and $|\lambda|\le bR$,
\begin{equation}\label{eq:flat}
 |\widetilde f_R(\lambda+i\rho,k)-1|\le\tfrac12
\end{equation}
for all sufficiently large $R$.  Moreover
$|\lambda+i\rho|/|\lambda+i3\rho|$ is bounded below for $\lambda\ge1$.
Consequently \eqref{eq:log} and \eqref{eq:flat} imply
\[
 \|\mathcal F_1f_R\|_{L^1(u\dd\mu)}\ge c\log R,
\]
whereas the proposed right side is
$C\|u\|_{1,\infty;\mu}\|f_R\|_1$, independent of $R$.  This proves
Theorem~\ref{thm:split}(1).

\section*{3. The strict interior is positive}

Fix $1<p<2$ and $p<q<p'$.  Without the harmless scalar factor, put
\[
 G_qf(\lambda)=
 \left(\int_K|\widetilde f(\lambda+i\rho_q,k)|^q\dd k\right)^{1/q}.
\]
Choose
\[
 p<r<\min\{q,q'\}.
\]
Then $1<r<2$ and $r<q<r'$.  The pointwise restriction estimate reproduced
as Theorem 2.3 in the source and its Hausdorff--Young estimate, Theorem 2.2,
give respectively
\[
 G_q:L^1(\X)\longrightarrow L^\infty(\mu),
 \qquad
 G_q:L^r(\X)\longrightarrow L^{r'}(\mu).
\]
The map $G_q$ is sublinear.  Real interpolation with
\[
 \frac1p=1-\theta+\frac\theta r
 \quad\Longrightarrow\quad
 \frac\theta{r'}=\frac1{p'}
\]
therefore yields the Lorentz refinement
\begin{equation}\label{eq:lorentzHY}
 \|G_qf\|_{L^{p',p}(\mu)}\le C_{p,q}\|f\|_{L^p(\X)}.
\end{equation}

Now $G_q^p\in L^{1/(p-1),1}(\mu)$, while the weak-weight hypothesis gives
\[
 u^{2-p}\in L^{1/(2-p),\infty}(\mu),
 \qquad
 \|u^{2-p}\|_{1/(2-p),\infty}
 =\|u\|_{1,\infty;\mu}^{2-p}.
\]
The two displayed exponents are conjugate.  Lorentz H\"older and
\eqref{eq:lorentzHY} show
\begin{align*}
 \int_\R G_qf(\lambda)^p u(\lambda)^{2-p}\dd\mu(\lambda)
 &\le C\|G_qf\|_{p',p}^p\|u\|_{1,\infty;\mu}^{2-p}\\
 &\le C_{p,q}\|f\|_p^p\|u\|_{1,\infty;\mu}^{2-p}.
\end{align*}
Finally, the scalar factor in $\mathcal F_q$ is at most one because
\[
 (\rho_q+2\rho)^2-\rho_q^2=4\rho(\rho_q+\rho)=\frac{8\rho^2}{q}>0.
\]
Taking $p$th roots proves \eqref{eq:positive}.

\section*{4. Exact remaining boundary}

The interpolation choice $p<r<\min\{q,q'\}$ exists exactly when
$p<q<p'$.  At $q=p$ or $q=p'$ it collapses, and the source's strong
Hausdorff--Young target $L^{p'}$ does not by itself imply the needed sharper
$L^{p',p}$ estimate.  No claim is made for those two edge lines when
$1<p<2$.  The unitary line $q=2$ is already covered by Theorem 1.9 of the
source.  Constants in the interior proof necessarily deteriorate as
$p\downarrow1$, consistently with the logarithmic endpoint counterexample.

\paragraph{Upgrade audit.}
Eight focused stages were completed: exact-scope and literature audits;
critical-weight scaling; approximate-identity construction; generalization
to all rank-one spaces; interior Lorentz interpolation; weak-weight pairing;
and edge-line analysis.  The literal question is settled negatively, and its
strict interior is settled positively.

\paragraph{Novelty boundary.}
Bounded primary-source searches through 17 August 2026 found no later paper
recording either conclusion.  The Lorentz argument is short and may be known
folklore; priority requires a broader search and expert review.
