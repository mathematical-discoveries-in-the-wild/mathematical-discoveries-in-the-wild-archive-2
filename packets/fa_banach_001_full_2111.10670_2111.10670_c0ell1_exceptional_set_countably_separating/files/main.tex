\section{Source and exact question}

The source is V.~dos Santos Ronchim and D.~V.~Tausk,
\emph{Extension of $c_0(I)$-valued operators on spaces of continuous
functions on compact lines}~\cite{RT}, arXiv:2111.10670, published in
\emph{Studia Mathematica} 268 (2023), 259--289.

For a compact line $K$, a subset $Q\subseteq K$, and a weak-star-null family
$(F_i)_{i\in I}$ in $\NBV(K)$, source Corollary 4.10 proves that
\[
 E:=\{t\in Q:(F_i(t))_{i\in I}\notin c_0(I)\}
\]
admitting a countably separating family implies that $(F_i)_{i\in I}$ is of
type $c_0\ell_1$ over $Q$. Immediately after the corollary, on printed page
27 (PDF page 27), the authors ask:

\begin{quote}
``We do not know if the converse of Corollary 4.10 holds.''

\section{Definitions}

For an arbitrary set $I$, the space $c_0(I)$ consists of the scalar families
$(x_i)_{i\in I}$ such that
\[
 \{i\in I:|x_i|\geq\varepsilon\}\quad\text{is finite for every }\varepsilon>0.
\]
The space $\NBV(K)$ is the source's right-continuous bounded-variation model
of $C(K)^*$.

A family $(F_i)_{i\in I}$ is \emph{of type $c_0\ell_1$ over $Q$} if there
are scalar arrays $(a_{i,t})$ and $(b_{i,t})$ such that
\[
 F_i(t)=a_{i,t}+b_{i,t},\qquad
 (a_{i,t})_{i\in I}\in c_0(I)\quad(t\in Q),
 \qquad
 \sup_{i\in I}\sum_{t\in Q}|b_{i,t}|<\infty.
\]

A family $(V_t)_{t\in E}$ is \emph{countably separating} for a subspace
$E\subseteq K$ if each $V_t$ is a neighborhood of $t$ in $E$ and every
$M\subseteq E$ satisfying
\[
 x\in V_y\qquad(x,y\in M)
\]
is countable.

We use the following finiteness lemma proved by Ronchim and Tausk immediately
after their open question.

\begin{lemma}[source Lemma 4.11]\label{lem:source}
Let $(F_i)_{i\in I}$ be weak-star-null in $\NBV(K)$, let $J\subseteq K$ be
infinite, and let $\delta>0$. Then
\[
 \{i\in I: |F_i(t)|\geq\delta\text{ for every }t\in J\}
\]
is finite.
\end{lemma}

\section{Proof intuition}

The $c_0$ part of a $c_0\ell_1$ decomposition cannot account for an
exceptional point, so at each exceptional $t$ one can select a coordinate
$i(t)$ from the $\ell_1$ part at which $F_{i(t)}(t)$ is bounded away from
zero. Each fixed coordinate is selected only countably often because an
absolutely summable row has countable support. Right continuity keeps the
selected function large immediately to the right of $t$.

Now suppose a set $M$ lies mutually in all the chosen neighborhoods. For a
fixed threshold, $F_{i(y)}$ is large on the entire right tail of $y$ in
$M$. Source Lemma 4.11 says that an infinite tail allows only finitely many
such coordinates. As $y$ moves to the right, the tails shrink, so these
finite coordinate sets form an inclusion chain. A chain of finite sets has
countable union. Countable selected-coordinate fibers then force the points
with infinite right tail to be countable; the points with finite right tail
are automatically countable. This is exactly the countable-separation
condition.

\section{The converse and the exact characterization}

\begin{theorem}\label{thm:main}
Let $K$ be a compact line, let $Q\subseteq K$, and let $(F_i)_{i\in I}$ be a
weak-star-null family in $\NBV(K)$. If $(F_i)_{i\in I}$ is of type
$c_0\ell_1$ over $Q$, then
\[
 E=\{t\in Q:(F_i(t))_{i\in I}\notin c_0(I)\}
\]
admits a countably separating family. Consequently,
\[
 (F_i)_{i\in I}\text{ is of type }c_0\ell_1\text{ over }Q
 \quad\Longleftrightarrow\quad
 E\text{ admits a countably separating family}.
\]
\end{theorem}

\begin{proof}
Assume that $(F_i)_{i\in I}$ is of type $c_0\ell_1$ over $Q$, and fix a
decomposition
\begin{equation}\label{eq:decomp}
 F_i(t)=a_{i,t}+b_{i,t}
\end{equation}
as in the definition. For each $i\in I$, the support
\[
 B_i:=\{t\in Q:b_{i,t}\neq0\}
\]
is countable, since $\sum_{t\in Q}|b_{i,t}|<\infty$.

Fix $t\in E$. Since $(F_i(t))_{i\in I}\notin c_0(I)$, there is an integer
$m(t)\geq1$ for which
\[
 \{i\in I:|F_i(t)|\geq 1/m(t)\}
\]
is infinite. On the other hand, $(a_{i,t})_{i\in I}\in c_0(I)$, so all but
finitely many indices in this infinite set satisfy
$|a_{i,t}|<1/(2m(t))$. Choose one such index and call it $i(t)$. By
\eqref{eq:decomp},
\[
 |b_{i(t),t}|\geq\frac{1}{2m(t)}>0.
\]
In particular,
\begin{equation}\label{eq:fiber}
 \{t\in E:i(t)=i\}\subseteq B_i
\end{equation}
is countable for every $i\in I$.

No point of $E$ is right-isolated in $K$. Indeed, if $t$ is right-isolated,
then $[\min K,t]$ is clopen, and weak-star nullity applied to its continuous
characteristic function gives $F_i(t)\to0$ in the $c_0(I)$ sense.

Thus right continuity of $F_{i(t)}$ at $t$ gives an open convex neighborhood
$U_t$ of $t$ in $K$ such that
\begin{equation}\label{eq:rightlarge}
 |F_{i(t)}(x)|\geq\frac{1}{2m(t)}
 \qquad\text{whenever }x\in U_t\text{ and }x\geq t.
\end{equation}
Set $V_t=U_t\cap E$. We prove that $(V_t)_{t\in E}$ is countably
separating.

Let $M\subseteq E$ satisfy $x\in V_y$ for all $x,y\in M$. Partition
\[
 M=\bigcup_{m\geq1}M_m,
 \qquad M_m:=\{t\in M:m(t)=m\}.
\]
Fix $m$. For $y\in M_m$, define its right tail in $M_m$ by
\[
 J_y:=\{x\in M_m:x\geq y\}.
\]
Since $M_m\subseteq M\subseteq V_y$ and $m(y)=m$, equation
\eqref{eq:rightlarge} shows that
\begin{equation}\label{eq:tailhigh}
 |F_{i(y)}(x)|\geq\frac1{2m}\qquad(x\in J_y).
\end{equation}

For each $y$ for which $J_y$ is infinite, put
\[
 S_y:=\left\{i\in I:|F_i(x)|\geq\frac1{2m}
                    \text{ for every }x\in J_y\right\}.
\]
Lemma~\ref{lem:source} makes $S_y$ finite, and \eqref{eq:tailhigh} gives
$i(y)\in S_y$. If $y\leq z$, then $J_z\subseteq J_y$, hence
$S_y\subseteq S_z$. Therefore the distinct sets $S_y$ form a chain of
finite sets under inclusion. Such a chain has at most one distinct member of
each finite cardinality, so
\[
 S:=\bigcup_{\substack{y\in M_m\\J_y\text{ infinite}}}S_y
\]
is countable. By \eqref{eq:fiber}, the set of all $y\in M_m$ with $J_y$
infinite is contained in a countable union of countable fibers and is
therefore countable.

The set of $y\in M_m$ with $J_y$ finite is also countable. In fact, if
$y<z$ and both tails are finite, then $J_z$ is a proper subset of $J_y$;
thus the map $y\mapsto |J_y|$ is injective on these points and takes values
in the positive integers. Hence $M_m$ is countable. Since
$M=\bigcup_{m\geq1}M_m$, the set $M$ is countable.

This proves that $(V_t)_{t\in E}$ is countably separating. The reverse
implication in the final equivalence is precisely source Corollary 4.10.
\end{proof}

\section{Verification notes and limitations}

\begin{enumerate}[leftmargin=*,itemsep=0.6em]
\item \textbf{One-sided continuity is sufficient.} The construction does not
claim that $F_{i(t)}$ is large to the left of $t$. In a clique, only the
right tail $J_t$ is used.

\item \textbf{Every exceptional point is eligible.} Right-isolated points
cannot belong to $E$, so right continuity always supplies the required
nontrivial right neighborhood.

\item \textbf{No cardinality assumption on $I$.} Countability enters only
through the support of each absolutely summable row $b_i$ and the countable
threshold partition indexed by $m$.

\item \textbf{The tail cases are exhaustive.} Infinite tails are controlled
by Lemma 4.11 and the nested finite-coordinate sets. Finite tails are
countable because their finite cardinalities strictly decrease along the
order.

\item \textbf{Operator consequence.} Together with source Theorem 2.7, the
result gives the topological characterization anticipated after Corollary
4.10 for extendible $c_0(I)$-valued operators in the compact-line setting.
It does not address extension problems outside that setting.

\item \textbf{Computation.} No numerical or finite computation is used or
needed; the verification is entirely deductive.
\end{enumerate}
