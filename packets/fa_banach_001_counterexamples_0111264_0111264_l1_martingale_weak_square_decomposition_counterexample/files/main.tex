\begin{abstract}
Problem 6.6 of Randrianantoanina asks for a weak-$L^1$ row/column
square-function estimate for every bounded $L^1$ martingale, with the
decomposition pieces required to belong to the strong Hardy spaces
$\mathcal H^1_C$ and $\mathcal H^1_R$.  The statement as printed is
false already in a commutative probability space.  We construct a positive,
uniformly integrable martingale $x$ with $\sup_n\|x_n\|_1<\infty$ but
$\|S(x)\|_1=\infty$.  In the commutative case both row and column Hardy
spaces equal $H^1$, so any admissible decomposition would force $x\in H^1$.
\end{abstract}

\section{The source problem}

Problem 6.6 asks whether there is a constant $K$ such that every bounded
$L^1$ martingale $x$ satisfies
\[
 \inf_{\substack{x=y+z\\y\in\Hc,\ z\in\Hr}}
 \bigl(\|dy\|_{L^{1,\infty}(\mathcal M;\ell^2_C)}
       +\|dz\|_{L^{1,\infty}(\mathcal M;\ell^2_R)}\bigr)
 \le K\|x\|_1.
\]
Here is the actual statement from source PDF page 29:

The strong memberships $y\in\Hc$ and $z\in\Hr$ are decisive.  We show
that for one bounded $L^1$ martingale there is no such pair at all.

\section{The martingale}

Let $(\Omega,\F,\mathbb P)$ be a nonatomic probability space partitioned
into pairwise disjoint sets $A_k$, $k\ge1$, with
$\mathbb P(A_k)=p_k=2^{-k}$.  Inside each $A_k$ choose nested sets
\[
 A_k=I_{k,0}\supset I_{k,1}\supset\cdots\supset I_{k,k},
 \qquad \mathbb P(I_{k,j})=p_k2^{-j},
\]
and put $J_{k,j}=I_{k,j-1}\setminus I_{k,j}$.  Thus
$\mathbb P(J_{k,j})=p_k2^{-j}$.

Set $a_k=k^{-2}$ and $b_k=a_k/p_k$.  Define the nonnegative terminal
variable
\[
 f=\sum_{k\ge1} b_k2^k\mathbf1_{I_{k,k}}.
\]
Its supports are disjoint, and hence
\[
 \|f\|_1=\sum_{k\ge1}b_k2^k\mathbb P(I_{k,k})
        =\sum_{k\ge1}a_k<\infty.
\]

Let $\F_1=\sigma(A_k:k\ge1)$.  For $n\ge1$, obtain $\F_{n+1}$ by
revealing, in every block $A_k$ with $k\ge n$, the next split
$I_{k,n}\subset I_{k,n-1}$; blocks whose depth has been exhausted remain
unchanged.  Equivalently, $\F_{n+1}$ is generated by all $A_k$ and all
$I_{k,j}$ with $1\le j\le\min\{n,k\}$.  Put
\[
 x_n=\E(f\mid\F_n),\qquad n\ge1.
\]
Then $x$ is a positive, uniformly integrable martingale and
\[
 \sup_n\|x_n\|_1=\|f\|_1=\sum_{k\ge1}k^{-2}.
\]

On $A_k$ its values are especially simple:
\[
 x_1=b_k\mathbf1_{A_k},\qquad
 x_{j+1}=b_k2^j\mathbf1_{I_{k,j}}\quad(1\le j\le k),
\]
and it is constant thereafter.  With the convention $x_0=0$, the local
difference at level $j$ is therefore
\[
 dx_{j+1}=
 \begin{cases}
  b_k2^{j-1},&\text{on }I_{k,j},\\
 -b_k2^{j-1},&\text{on }J_{k,j},\\
 0,&\text{on }A_k\setminus I_{k,j-1}.
 \end{cases}
\]

\section{Divergence of the square function}

\begin{theorem}\label{thm:counterexample}
The martingale $x$ is bounded in $L^1$, but its ordinary square function
\[
 S(x)=\left(\sum_{n\ge1}|dx_n|^2\right)^{1/2}
\]
does not belong to $L^1$.
\end{theorem}

\begin{proof}
Fix $k\ge1$ and $1\le j\le k$.  On $J_{k,j}$ the square function
dominates the absolute value of the $(j+1)$st difference, so
\[
 S(x)\ge b_k2^{j-1}\quad\text{on }J_{k,j}.
\]
Consequently,
\[
 \int_{J_{k,j}}S(x)\,d\mathbb P
 \ge b_k2^{j-1}p_k2^{-j}=\frac{a_k}{2}.
\]
The sets $J_{k,j}$ are mutually disjoint over all $(k,j)$.  Summing gives
\[
 \|S(x)\|_1
 \ge\frac12\sum_{k\ge1}\sum_{j=1}^k a_k
 =\frac12\sum_{k\ge1}\frac1k=\infty.
\]
\end{proof}

\section{Negative answer to Problem 6.6}

\begin{theorem}
There is no universal constant in Problem 6.6.  Indeed, for the martingale
$x$ above, the set of decompositions over which the displayed infimum is
taken is empty.
\end{theorem}

\begin{proof}
View the probability space as the abelian finite von Neumann algebra
$\mathcal M=L^\infty(\Omega)$.  In an abelian algebra the column and row
square functions coincide with $S$.  Suppose that
$x=y+z$ with $y\in\Hc$ and $z\in\Hr$.  Pointwise Minkowski's inequality
in $\ell^2$ gives
\[
 S(x)=\left(\sum_n|dy_n+dz_n|^2\right)^{1/2}
 \le S(y)+S(z).
\]
The right-hand side is integrable by the two strong Hardy memberships.
It follows that $S(x)\in L^1$, contrary to
Theorem~\ref{thm:counterexample}.  No admissible pair $(y,z)$ exists, so
the infimum is $+\infty$ and cannot be bounded by $K\|x\|_1$.
\end{proof}

\section{Relation to later dense-subcase results}

Later papers proved weak row/column square-function decompositions for
$L^2$ martingales with an $L^1$ right-hand side.  That does not contradict
the example: the terminal variable above need not be in $L^2$, and the
strong Hardy-membership requirement in the universal $L^1$ formulation
already makes the admissible class empty.  A viable endpoint question would
have to enlarge the decomposition class to a weak Hardy completion (or
restrict the input class), rather than require both pieces to lie in strong
$H^1$.
