\title{The probability-level lifting map is always surjective\\
\large An affirmative answer to the open question in arXiv:2402.04926}
\author{Autonomous solve-first investigation, lane 12}

\begin{abstract}
Oberguggenberger asks whether the probability-level evaluation map
$\kappa_1$ for algebraic-dual-valued random functionals is surjective for
every locally convex space.  It is.  The unrestricted map $\kappa_0$ is
surjective by a Hamel-basis lifting, and every $\kappa_0$-lift of a
sequentially continuous map $E\to L^0(\Omega)$ is automatically continuous
in probability.  Thus restricting both domain and codomain to the
probability-continuous level preserves surjectivity.  The proof works for
every probability space and every Hausdorff locally convex space, with no
nuclearity, separability, completeness, or regularity assumption.
\end{abstract}

\section{The question}

Fix a probability space $(\Om,\Sigma,P)$ and a Hausdorff locally convex
space $E$.  Give the algebraic dual $\Ealg$ the cylinder sigma-algebra.
Write $\Lzero(\Om)$ for real measurable functions modulo almost-sure
equality, with its usual topology of convergence in probability.

For a cylinder-measurable map $X:\Om\to\Ealg$, define
\[
  (\kappa_0X)(e)
  =\bigl[\omega\longmapsto X(\omega)(e)\bigr],\qquad e\in E.
\]
The paper writes $\Lzero(\Om:\Ealg)_{\rm prob}$ for those $X$ such that,
whenever $e_n\to e$ in $E$,
\[
     X(\mathord\cdot)(e_n)\longrightarrow X(\mathord\cdot)(e)
     \quad\hbox{in probability}.
\]
Its map $\kappa_1$ is the restriction
\begin{equation}\label{eq:kappa1}
 \kappa_1:\Lzero(\Om:\Ealg)_{\rm prob}
 \longrightarrow \Lin(E:\Lzero(\Om))_{\rm prob}
 =\Lc(E:\Lzero(\Om)).
\end{equation}
At the end of Section 7, the paper asks whether \eqref{eq:kappa1} is
surjective in general \cite[p.~17]{Oberguggenberger}.  The exact passage is
reproduced in Figure~\ref{fig:source}.

\section{A general affirmative answer}

\begin{theorem}\label{thm:main}
For every probability space $(\Om,\Sigma,P)$ and every Hausdorff locally
convex space $E$, the map $\kappa_1$ in \eqref{eq:kappa1} is surjective.
\end{theorem}

\begin{proof}
Let $Y\in\Lc(E:\Lzero(\Om))$.  Choose a Hamel basis
$(u_\lambda)_{\lambda\in\Lambda}$ of $E$.  For each $\lambda$, choose a
measurable representative $y_\lambda:\Om\to\R$ of the class
$Y(u_\lambda)\in\Lzero(\Om)$.

Every $e\in E$ has a unique finite expansion
$e=\sum_{\lambda\in F}a_\lambda u_\lambda$.  Define
\begin{equation}\label{eq:lift}
 X(\omega)(e)=\sum_{\lambda\in F}a_\lambda y_\lambda(\omega).
\end{equation}
For every $\omega$, the map $e\mapsto X(\omega)(e)$ is linear, so
$X(\omega)\in\Ealg$.  For every fixed $e$, the right side of
\eqref{eq:lift} is a finite sum of measurable functions.  Hence
$X:\Om\to\Ealg$ is measurable for the cylinder sigma-algebra.

Linearity of $Y$ in $\Lzero(\Om)$ gives, for every finite expansion,
\[
 [X(\mathord\cdot)(e)]
 =\sum_{\lambda\in F}a_\lambda [y_\lambda]
 =\sum_{\lambda\in F}a_\lambda Y(u_\lambda)
 =Y(e).
\]
Thus $\kappa_0X=Y$.

It remains only to check that this unrestricted lift lies in the domain of
$\kappa_1$.  If $e_n\to e$ in $E$, sequential continuity of $Y$ gives
\[
 Y(e_n)\longrightarrow Y(e)\quad\hbox{in }\Lzero(\Om),
\]
which is precisely convergence in probability.  Since
$[X(\mathord\cdot)(e_n)]=Y(e_n)$ and
$[X(\mathord\cdot)(e)]=Y(e)$, we obtain
\[
 X(\mathord\cdot)(e_n)\longrightarrow X(\mathord\cdot)(e)
 \quad\hbox{in probability}.
\]
Therefore $X\in\Lzero(\Om:\Ealg)_{\rm prob}$, and on this domain
$\kappa_1X=\kappa_0X=Y$.
\end{proof}

The mechanism can be summarized by the exact identity
\begin{equation}\label{eq:preimage}
 \Lzero(\Om:\Ealg)_{\rm prob}
 =\kappa_0^{-1}\!\left(\Lin(E:\Lzero(\Om))_{\rm prob}\right).
\end{equation}
Consequently, surjectivity of $\kappa_0$ onto the unrestricted right-hand
space implies surjectivity of its restriction onto the probability level.

\begin{lemma}[null-set audit]\label{lem:null}
No common exceptional set is needed in the final continuity step.
\end{lemma}

\begin{proof}
Convergence in probability depends only on the classes in $\Lzero(\Om)$.
For any fixed convergent sequence $(e_n)$, the equalities
$[X(\mathord\cdot)(e_n)]=Y(e_n)$ involve only countably many coordinates;
changing any representative on a null set does not change their convergence
in probability.  The pathwise common-null-set requirement belongs to the
strictly stronger ``path'' level and is absent from the definition of the
domain in \eqref{eq:kappa1}.
\end{proof}

\begin{corollary}
The open question quoted in Figure~\ref{fig:source} has an affirmative
answer, without assuming property $(\mathcal R)$.
\end{corollary}

\paragraph{Scope.}
Theorem~\ref{thm:main} settles surjectivity of $\kappa_1$.  It makes no claim
that $\kappa_1$ is injective, nor that the maps involving the sequentially
continuous dual $E'$ are surjective; the counterexamples to those stronger
properties in the source remain untouched.
