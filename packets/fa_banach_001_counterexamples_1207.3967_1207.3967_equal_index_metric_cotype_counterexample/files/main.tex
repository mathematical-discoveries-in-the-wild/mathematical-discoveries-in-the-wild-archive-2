\section{Source question}

Kraus studies coarse and uniform embeddings between Orlicz sequence spaces
$h_M$ and $h_N$ \cite{Kraus}.  On page 11, after treating all other index
configurations, the paper asks whether there exist Orlicz functions $M,N$ such
that
\[
     2<\beta_M=\beta_N<\infty
\]
and $h_M$ does not coarsely (uniformly) embed into $h_N$.

\paragraph{Transcription.}
``However, we do not know whether there exist Orlicz functions $M,N$
satisfying $2<\beta_M=\beta_N<\infty$, such that $h_M$ does not coarsely
(uniformly) embed into $h_N$.''

\section{Definitions and the metric-cotype input}

An Orlicz function is a continuous, nondecreasing convex function
$M:[0,\infty)\to[0,\infty)$ with $M(0)=0$ and $M(t)\to\infty$.  Its sequence
space $h_M$ is equipped with the Luxemburg norm
\[
 \norm{x}_{M}=\inf\left\{\rho>0:
       \sum_i M\!\left(\frac{|x_i|}{\rho}\right)\le 1\right\}.
\]
For the present normalization, the upper Matuszewska--Orlicz index is
\[
 \beta_M=\inf\left\{q:\inf_{\lambda,t\in(0,1]}
 \frac{M(\lambda t)}{M(\lambda)t^q}>0\right\}.
\]

We use the following specialization of the optimal-scale metric-cotype
theorem of Mendel and Naor \cite[Theorem 4.1]{MN}.  It is important here that
$\ell_p$ is $K$-convex and has cotype $p$.

\begin{lemma}[Metric cotype of $\ell_p$]\label{lem:mc}
For every $p>2$ there are constants $\Gamma,C<\infty$ such that for each
$n\ge1$ one can choose an integer $m=m_n$, divisible by four, satisfying
\[
 C^{-1}n^{1/p}\le m\le Cn^{1/p},
\]
and every map $f:\Z_m^n\to\ell_p$ satisfies
\begin{equation}\label{eq:mc}
 \sum_{j=1}^n \E_x
 \norm{f(x+\tfrac m2e_j)-f(x)}_p^p
 \le
 \Gamma^p m^p\E_{x,\varepsilon}
 \norm{f(x+\varepsilon)-f(x)}_p^p,
\end{equation}
where $x$ is uniform on $\Z_m^n$ and
$\varepsilon$ is uniform on $\{-1,0,1\}^n$.
\end{lemma}

\begin{proof}
Apply Mendel--Naor's theorem with both of its exponents equal to $p$ and
$X=\ell_p$.  It gives a constant threshold $A n^{1/p}$ and a uniform
metric-cotype constant for every multiple of four above that threshold.
Choose the least such multiple above $\max\{4,A n^{1/p}\}$.  This choice is
also at most a constant times $n^{1/p}$.
\end{proof}

\section{The counterexample}

Fix $p>2$ and define
\begin{equation}\label{eq:M}
 M(0)=0,\qquad
 M(t)=\frac{t^p}{1-\log t}\quad(0<t\le1),\qquad
 M(t)=1+(p+1)(t-1)\quad(t\ge1).
\end{equation}
Let $N(t)=t^p$.

\begin{theorem}\label{thm:main}
For every $p>2$, the functions $M,N$ in \eqref{eq:M} satisfy
\[
       \alpha_M=\beta_M=\beta_N=p,
\]
but $h_M$ admits neither a coarse embedding nor a uniform embedding into
$h_N=\ell_p$.  Consequently the page-11 question of Kraus has an affirmative
answer for every possible common value $p>2$.
\end{theorem}

\section{Proof intuition}

The logarithmic denominator leaves both Matuszewska--Orlicz indices equal to
$p$, but makes the fundamental function of the canonical basis slightly
smaller than that of $\ell_p$:
\[
 \Phi_M(n)=\norm{e_1+\cdots+e_n}_M=o(n^{1/p}).
\]
To turn this endpoint defect into a nonlinear obstruction, place $n$ circles
inside the first $2n$ coordinates of $h_M$.  Moving one coordinate halfway
around its circle has fixed size, while moving every coordinate by at most one
torus step has size at most $\Phi_M(2n)/m_n=o(1)$.  The metric-cotype
inequality says that these two scales cannot be separated in this way after a
map into $\ell_p$.  Scaling the circles so short steps have size one defeats a
coarse embedding; leaving them unscaled defeats a uniform embedding.

\section{Proof of Theorem \ref{thm:main}}

\subsection*{Step 1: the Orlicz function and its indices}

Put $L(t)=1-\log t$.  On $(0,1]$ a direct differentiation gives
\[
 M'(t)=t^{p-1}\left(\frac{p}{L(t)}+\frac{1}{L(t)^2}\right)>0,
 \qquad
 M''(t)=\frac{t^{p-2}}{L(t)^3}
 \left[p(p-1)L(t)^2+(2p-1)L(t)+2\right]>0.
\]
Also $M'(1^-)=p+1$, so the tangent linear continuation in
\eqref{eq:M} is convex.  Thus $M$ is an Orlicz function.

For $\lambda,t\in(0,1]$,
\begin{equation}\label{eq:indexratio}
 \frac{M(\lambda t)}{M(\lambda)t^q}
 =t^{p-q}\frac{L(\lambda)}{L(\lambda t)}.
\end{equation}
If $q\le p$, the right side is at most $t^{p-q}\le1$, which yields
$\alpha_M\ge p$.  If $q>p$, then $L(\lambda)\ge1$ and
\[
 t^{p-q}\frac{L(\lambda)}{L(\lambda t)}
 =\frac{t^{p-q}}{1+(-\log t)/L(\lambda)}
 \ge \frac{t^{p-q}}{1-\log t}.
\]
The last function is positive and continuous on $(0,1]$ and tends to
$\infty$ at zero, so it has a positive infimum.  Hence $\beta_M\le p$.
Since always $\alpha_M\le\beta_M$, we have
$\alpha_M=\beta_M=p$.  Clearly $\beta_N=p$ and $h_N=\ell_p$.

Let
\[
 \Phi(k)=\norm{e_1+\cdots+e_k}_M=\frac{1}{M^{-1}(1/k)}.
\]
Writing $s_k=M^{-1}(1/k)$, we have
$k^{-1}=s_k^p/L(s_k)$ and therefore
\begin{equation}\label{eq:fundamental}
 \frac{\Phi(k)}{k^{1/p}}
 =\frac{1}{L(s_k)^{1/p}}\longrightarrow0.
\end{equation}

\subsection*{Step 2: a family of discrete tori in $h_M$}

Let $m=m_n$ be supplied by Lemma \ref{lem:mc}.  For $a>0$ define
$G_{n,a}:\Z_m^n\to h_M$ by
\[
 G_{n,a}(x)=a\sum_{j=1}^n\left[
 \cos\!\left(\frac{2\pi x_j}{m}\right)e_{2j-1}
 +\sin\!\left(\frac{2\pi x_j}{m}\right)e_{2j}\right].
\]
Shifting $x_j$ by $m/2$ changes the corresponding two-coordinate vector to
its negative.  Since one of the absolute values of its sine and cosine is at
least $1/\sqrt2$, lattice monotonicity of the Luxemburg norm gives
\begin{equation}\label{eq:long}
 \norm{G_{n,a}(x+\tfrac m2e_j)-G_{n,a}(x)}_M\ge\sqrt2\,a.
\end{equation}
The sine and cosine functions are one-Lipschitz.  Hence, for every
$\varepsilon\in\{-1,0,1\}^n$, each of the first $2n$ coordinates of
$G_{n,a}(x+\varepsilon)-G_{n,a}(x)$ has absolute value at most
$2\pi a/m$.  Again by lattice monotonicity and symmetry,
\begin{equation}\label{eq:short}
 \norm{G_{n,a}(x+\varepsilon)-G_{n,a}(x)}_M
 \le \frac{2\pi a}{m}\Phi(2n).
\end{equation}

\subsection*{Step 3: no coarse embedding}

Suppose $F:h_M\to\ell_p$ were a coarse embedding.  Let
$\rho:[0,\infty)\to[0,\infty)$ be a nondecreasing lower control with
$\rho(t)\to\infty$, and let
\[
 W=\sup\{\norm{F(u)-F(v)}_p:\norm{u-v}_M\le1\}<\infty.
\]
Set
\[
 a_n=\frac{m_n}{2\pi\Phi(2n)}.
\]
By \eqref{eq:fundamental} and $m_n\asymp n^{1/p}$, we have
$a_n\to\infty$.  Equations \eqref{eq:long}--\eqref{eq:short} and
Lemma \ref{lem:mc}, applied to $F\circ G_{n,a_n}$, give
\[
 n\rho(\sqrt2a_n)^p
 \le \Gamma^p m_n^p W^p.
\]
Thus
\[
 \rho(\sqrt2a_n)
 \le \Gamma\frac{m_n}{n^{1/p}}W\le C\Gamma W,
\]
contradicting $a_n\to\infty$ and $\rho(t)\to\infty$.

\subsection*{Step 4: no uniform embedding}

Suppose instead that $F:h_M\to\ell_p$ were a uniform embedding.  Define
\[
 \omega_F(t)=\sup\{\norm{F(u)-F(v)}_p:\norm{u-v}_M\le t\}.
\]
Uniform continuity gives $\omega_F(t)\to0$ as $t\downarrow0$ (and makes
$\omega_F(t)$ finite).  Uniform continuity of $F^{-1}$ on $F(h_M)$ gives an
$\eta>0$ such that
\begin{equation}\label{eq:inverse}
 \norm{u-v}_M\ge\sqrt2
 \quad\Longrightarrow\quad
 \norm{F(u)-F(v)}_p\ge\eta.
\end{equation}
Take $a=1$ in $G_{n,a}$.  By \eqref{eq:fundamental},
\[
 \delta_n=\frac{2\pi\Phi(2n)}{m_n}\longrightarrow0.
\]
Applying Lemma \ref{lem:mc} to $F\circ G_{n,1}$ and using
\eqref{eq:long}, \eqref{eq:short}, and \eqref{eq:inverse}, we obtain
\[
 n\eta^p\le \Gamma^p m_n^p\omega_F(\delta_n)^p.
\]
Consequently
\[
 \eta\le \Gamma\frac{m_n}{n^{1/p}}\omega_F(\delta_n)\longrightarrow0,
\]
a contradiction.  This completes the proof. \qed

\section{Verification and limitations}

\begin{itemize}
\item The Orlicz convexity and index computations are explicit.  The same
logarithmic function was used by Kraus at the Hilbert endpoint $p=2$; here it
is used for every $p>2$.
\item The only external mathematical input is the optimal $m_n\asymp n^{1/p}$
metric-cotype inequality for $\ell_p$, obtained from
\cite[Theorem 4.1]{MN}.  Merely knowing the cotype exponent would not suffice.
\item No numerical experiment is used.  The proof is scale- and
constant-robust.
\item A bounded novelty check on 9 August 2026 searched the exact source
sentence, the paper title and arXiv id, and combinations of ``Orlicz sequence
space'', ``coarse embedding'', ``uniform embedding'', equal upper index, and
Matuszewska--Orlicz.  It located the source and general later embedding work,
but no later paper claiming this endpoint answer.  This is not a proof of
novelty.
\item Human review should focus on the specialization of Mendel--Naor's
Theorem 4.1 and on the two modulus arguments after
\eqref{eq:long}--\eqref{eq:short}.
\end{itemize}
