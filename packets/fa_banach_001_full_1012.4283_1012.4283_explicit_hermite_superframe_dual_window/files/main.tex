\begin{center}
\fcolorbox{accent}{soft}{\parbox{0.91\linewidth}{
\textbf{Result.}  Let
$\mathbf h=(H_0,\ldots,H_n)$ be the vector of the first $n+1$ normalized
Hermite functions, and let $\Lambda$ be a lattice with
$s(\Lambda)<1/(n+1)$.  A dual vector window
$\boldsymbol\gamma=(\gamma_0,\ldots,\gamma_n)$ is given explicitly below.
Its ordinary Bargmann transforms are a modified Weierstrass sigma function
times finite Taylor polynomials.  Every component belongs to the
Feichtinger algebra $M^1$, the full matrix Wexler--Raz identities hold, and
the resulting superframe reconstruction transfers to the vector modulation
spaces and to an explicit polyanalytic sampling kernel.}}
\end{center}

\section{The source remark}

Abreu asks for an explicit dual vectorial window after deriving a sampling
formula by decomposing the full polyanalytic Fock space into its true
polyanalytic summands \cite{Abreu}.  The earlier Hermite-superframe paper
reduces the dual-window problem to a finite Hermite interpolation problem
and gives a triangular ansatz, but leaves its coefficients implicit
\cite{GL}.  The formula below solves that triangular system in closed form.

\section{Notation and formula}

Put $N=n+1$.  Let $\Lambda^\circ$ be the adjoint lattice and
$\Omega=\overline{\Lambda^\circ}$.  Let $\sigma_\Omega$ be the modified
Weierstrass sigma function used in \cite{GL}; it has simple zeros precisely
at $\Omega$ and satisfies the standard lattice growth estimate.  Define
\[
 S(z)=\sigma_\Omega(z)^N,
 \qquad
 q(z)=\frac{S(z)}{z^N},
\]
where the singularity of $q$ at zero is removed.  Thus $q(0)\ne0$, and $q$
has a zero of order $N$ at every point of $\Omega\setminus\{0\}$.

For a function $a$ holomorphic near zero, write
\[
 [a]_r(z)=\sum_{k=0}^{r}\frac{a^{(k)}(0)}{k!}z^k
\]
for its degree-$r$ Taylor truncation.  For $0\le j<N$, set
\begin{equation}\label{eq:Ej}
 E_j(z)=\frac{z^j}{j!}\,q(z)\,[q^{-1}]_{N-1-j}(z),
\end{equation}
and
\begin{equation}\label{eq:Gj}
 G_j(z)=s(\Lambda)\sqrt{\pi^j j!}\,E_j(z),
 \qquad
 \gamma_j=\Barg^{-1}G_j.
\end{equation}
Here $\Barg$ is the ordinary unitary Bargmann transform, with the conventions
of \cite{GL}.  Equations \eqref{eq:Ej}--\eqref{eq:Gj} are the promised
explicit dual window.

For example, when $N=2$ and
$q(z)=q_0+q_1z+O(z^2)$,
\[
 G_0=s(\Lambda)q(z)\left(\frac1{q_0}-\frac{q_1}{q_0^2}z\right),
 \qquad
 G_1=s(\Lambda)\sqrt\pi\,\frac{zq(z)}{q_0}.
\]

\section{Proof}

\begin{lemma}[Cardinal jets]\label{lem:jets}
For $0\le j,\ell<N$,
\begin{align*}
 E_j^{(\ell)}(0)&=\delta_{j\ell},\\
 E_j^{(\ell)}(\omega)&=0
   &&(\omega\in\Omega\setminus\{0\}).
\end{align*}
\end{lemma}

\begin{proof}
Taylor multiplication gives
\[
 q(z)[q^{-1}]_{N-1-j}(z)=1+O(z^{N-j}).
\]
Consequently
$E_j(z)=z^j/j!+O(z^N)$, which is exactly the first assertion.
At a nonzero $\omega\in\Omega$, the function $q$ has a zero of order $N$;
the remaining factors in \eqref{eq:Ej} are holomorphic there.  Hence $E_j$
also has a zero of order at least $N$, proving the second assertion.
\end{proof}

\begin{lemma}[Feichtinger regularity]\label{lem:m1}
Each $\gamma_j$ belongs to $M^1(\R)$.
\end{lemma}

\begin{proof}
The modified sigma estimate from \cite{GL} is
\[
 |S(z)|\lesssim
 \exp\!\left(\frac{\pi Ns(\Lambda)}2|z|^2\right).
\]
Outside a fixed disk, \eqref{eq:Ej} is a finite sum of terms
$c_kS(z)z^{j+k-N}$ with $j+k\le N-1$; thus it has at most a harmless
polynomial factor times the displayed Gaussian.  On the disk it is entire
and bounded.  Since $Ns(\Lambda)<1$,
\[
 \int_{\C}|G_j(z)|e^{-\pi|z|^2/2}\,dm(z)<\infty.
\]
The Bargmann characterization of the Feichtinger algebra therefore gives
$\gamma_j\in M^1(\R)$.
\end{proof}

\begin{theorem}[Explicit Hermite superframe dual]\label{thm:dual}
The vector $\boldsymbol\gamma=(\gamma_0,\ldots,\gamma_{N-1})$ defined by
\eqref{eq:Ej}--\eqref{eq:Gj} is a dual window for
$\mathcal G(\mathbf h,\Lambda)$ in $L^2(\R,\C^N)$.
\end{theorem}

\begin{proof}
For $\mu\in\Lambda^\circ$ and $0\le j,\ell<N$, the Bargmann--Hermite
identity in \cite{GL} reads
\begin{align}\label{eq:bh}
 \inner{\gamma_j}{\pi_\mu H_\ell}
  ={}&e^{-i\pi\mu_1\mu_2}e^{-\pi|\mu|^2/2}
  \frac1{\sqrt{\pi^\ell\ell!}}\notag\\
  &\times\sum_{k=0}^{\ell}\binom{\ell}{k}(-\pi\mu)^k
       G_j^{(\ell-k)}(\overline\mu).
\end{align}
If $\mu\ne0$, then $\overline\mu\in\Omega\setminus\{0\}$, and every
derivative appearing in \eqref{eq:bh} vanishes by Lemma \ref{lem:jets}.
If $\mu=0$, only the $k=0$ term remains and Lemma \ref{lem:jets} gives
\[
 \inner{\gamma_j}{H_\ell}
 =\frac{G_j^{(\ell)}(0)}{\sqrt{\pi^\ell\ell!}}
 =s(\Lambda)\delta_{j\ell}.
\]
Therefore
\[
 \frac1{s(\Lambda)}
 \inner{\gamma_j}{\pi_\mu H_\ell}
 =\delta_{\mu0}\delta_{j\ell},
\]
which is the full matrix Wexler--Raz condition.  Lemma \ref{lem:m1}
ensures that all component Gabor systems are Bessel.  The vector
Wexler--Raz theorem now proves that $\boldsymbol\gamma$ is a dual window.
\end{proof}

\section{Explicit superframe sampling kernel}

Let $\Barg^{j+1}$ denote the true polyanalytic Bargmann transform associated
with $H_j$.  Write
\[
 \mathbf B^N(f_0,\ldots,f_{N-1})
 =\sum_{j=0}^{N-1}\Barg^{j+1}f_j.
\]
Define
\begin{equation}\label{eq:Psi}
 \Psi_\Lambda^N(z)
 =\sum_{j=0}^{N-1}\Barg^{j+1}\gamma_j(z)
 =\sum_{j=0}^{N-1}
 \left(\frac{\pi^j}{j!}\right)^{1/2}
 e^{\pi|z|^2}\left(\frac d{dz}\right)^j
 \left[e^{-\pi|z|^2}G_j(z)\right].
\end{equation}
This is a finite expression in the modified sigma function, its reciprocal
Taylor coefficients at zero, and derivatives.

\begin{corollary}[Superframe reconstruction]
If $F=\mathbf B^N\mathbf f$, then
\begin{equation}\label{eq:sampling}
 F(z)=\sum_{\lambda\in\Lambda}
 F(\lambda)e^{\pi\overline\lambda z-\pi|\lambda|^2}
 \Psi_\Lambda^N(z-\lambda).
\end{equation}
For $1\le p<\infty$, the corresponding vector Gabor expansion converges in
the coordinatewise vector modulation space $M^p(\R,\C^N)$; for $p=\infty$
it converges weak-$^*$.
\end{corollary}

\begin{proof}
Theorem \ref{thm:dual} gives
\[
 \mathbf f=\sum_{\lambda\in\Lambda}
 \inner{\mathbf f}{\pi_\lambda\mathbf h}\,
 \pi_\lambda\boldsymbol\gamma.
\]
The polyanalytic Bargmann transform converts the coefficient into the
weighted sample of $F$ and intertwines time--frequency shifts with
Bargmann--Fock shifts.  Applying $\mathbf B^N$ yields
\eqref{eq:sampling} and \eqref{eq:Psi}.  Since all $H_j$ and $\gamma_j$
belong to $M^1$ and there are only finitely many components, the usual
bounded analysis and synthesis maps on modulation spaces give norm
convergence for finite $p$ and weak-$^*$ convergence at the endpoint.
\end{proof}

\section{What is new and what is inherited}

The density threshold, sigma-function growth estimate, Bargmann--Hermite
identity, vector Wexler--Raz theorem, and existence of a solution via a
triangular linear system all come from \cite{GL}.  The new step is the closed
formula \eqref{eq:Ej}: multiplication by a finite Taylor truncation of
$1/q$ solves all $N^2$ jet conditions simultaneously.  It converts the
implicit triangular construction into an explicit vector window and hence
supplies the superframe reconstruction that arXiv:1012.4283 called
non-explicit.  This is not a formula for the canonical dual; it is one
explicit dual with Gaussian-type phase-space decay.
